#!/bin/bash
set -e

echo "🚀 Initializing Miya AI Native Android Project..."

APP_JAVA=miya-app/app/src/main/java/com/miya/app

# NOTE: ab "rm -rf miya-app" nahi hota — pehle ye MiyaService.java ko bhi mita deta tha.
mkdir -p "$APP_JAVA"
mkdir -p miya-app/app/src/main/res/values
mkdir -p miya-app/app/src/main/assets

# MiyaService.java is script se generate nahi hoti — ye pehle se yahin honi chahiye
if [ ! -f "$APP_JAVA/MiyaService.java" ]; then
    echo "❌ $APP_JAVA/MiyaService.java nahi mili. Pehle ise us path par rakho, phir script chalao."
    exit 1
fi

# Purane generated assets saaf karke dobara copy karo
rm -rf miya-app/app/src/main/assets/*
if [ -d "assets" ]; then
    cp -r assets/* miya-app/app/src/main/assets/
    echo "✅ Assets copied successfully"
else
    echo "⚠️ No assets folder found"
fi

# index.html repo ke root me hai, app ko assets me chahiye
if [ -f "index.html" ]; then
    cp index.html miya-app/app/src/main/assets/index.html
    echo "✅ index.html copied to assets"
elif [ ! -f "miya-app/app/src/main/assets/index.html" ]; then
    echo "❌ index.html nahi mili (na root me, na assets me) — app blank khulega."
    exit 1
fi

# ---------------- settings.gradle ----------------
cat > miya-app/settings.gradle <<'EOF'
pluginManagement {
    repositories { google(); mavenCentral(); gradlePluginPortal() }
}
dependencyResolutionManagement {
    repositoriesMode.set(RepositoriesMode.FAIL_ON_PROJECT_REPOS)
    repositories { google(); mavenCentral() }
}

package com.miya.app;

import android.Manifest;
import android.app.Activity;
import android.content.Intent;
import android.content.pm.PackageManager;
import android.net.Uri;
import android.os.BatteryManager;
import android.os.Build;
import android.os.Bundle;
import android.os.Environment;
import android.provider.Settings;
import android.speech.RecognitionListener;
import android.speech.RecognizerIntent;
import android.speech.SpeechRecognizer;
import android.util.Base64;
import android.webkit.JavascriptInterface;
import android.webkit.PermissionRequest;
import android.webkit.WebChromeClient;
import android.webkit.WebResourceRequest;
import android.webkit.WebResourceResponse;
import android.webkit.WebSettings;
import android.webkit.WebView;
import android.webkit.WebViewClient;

import androidx.webkit.WebViewAssetLoader;

import org.json.JSONObject;

import java.io.File;
import java.io.FileOutputStream;
import java.util.ArrayList;

public class MainActivity extends Activity {
    private static final int REQ_PERMS = 100;
    private static final String HOST = "appassets.androidplatform.net";

    private WebView webView;
    private SpeechRecognizer nativeRec;
    private boolean overlayAsked = false;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        requestMissingPermissions();

        webView = new WebView(this);
        setContentView(webView);

        WebSettings ws = webView.getSettings();
        ws.setJavaScriptEnabled(true);
        ws.setDomStorageEnabled(true);
        ws.setAllowFileAccess(false);
        ws.setAllowContentAccess(false);
        ws.setMediaPlaybackRequiresUserGesture(false);

        // index.html ko https://appassets.androidplatform.net se serve karo (secure origin:
        // camera/mic, localStorage sab sahi chalte hain, file:// ki dikkatein nahi)
        final WebViewAssetLoader loader = new WebViewAssetLoader.Builder()
                .setDomain(HOST)
                .addPathHandler("/assets/", new WebViewAssetLoader.AssetsPathHandler(this))
                .build();

        webView.addJavascriptInterface(new MiyaBridge(), "Android");

        webView.setWebViewClient(new WebViewClient() {
            @Override
            public WebResourceResponse shouldInterceptRequest(WebView v, WebResourceRequest r) {
                return loader.shouldInterceptRequest(r.getUrl());
            }

            @Override
            public boolean shouldOverrideUrlLoading(WebView v, WebResourceRequest r) {
                // Sirf apne assets; baaki sab block (bridge baahar ke page par kabhi expose nahi hoga)
                return !HOST.equals(r.getUrl().getHost());
            }
        });

        webView.setWebChromeClient(new WebChromeClient() {
            @Override
            public void onPermissionRequest(final PermissionRequest request) {
                runOnUiThread(() -> {
                    if (!HOST.equals(request.getOrigin().getHost())) { request.deny(); return; }
                    ArrayList<String> ok = new ArrayList<>();
                    for (String res : request.getResources()) {
                        if (PermissionRequest.RESOURCE_VIDEO_CAPTURE.equals(res) && has(Manifest.permission.CAMERA)) {
                            ok.add(res);
                        } else if (PermissionRequest.RESOURCE_AUDIO_CAPTURE.equals(res) && has(Manifest.permission.RECORD_AUDIO)) {
                            ok.add(res);
                        }
                    }
                    if (ok.isEmpty()) request.deny();
                    else request.grant(ok.toArray(new String[0]));
                });
            }
        });

        webView.loadUrl("https://" + HOST + "/assets/index.html");
    }

    @Override
    protected void onResume() {
        super.onResume();
        startMiyaIfReady();
    }

    @Override
    public void onRequestPermissionsResult(int code, String[] perms, int[] results) {
        super.onRequestPermissionsResult(code, perms, results);
        startMiyaIfReady();
    }

    private boolean has(String perm) {
        return checkSelfPermission(perm) == PackageManager.PERMISSION_GRANTED;
    }

    // Jo permissions nahi mili unhe ek hi baar maango (pehle loop me baar-baar maang raha tha)
    private void requestMissingPermissions() {
        ArrayList<String> need = new ArrayList<>();
        String[] wanted = {Manifest.permission.RECORD_AUDIO, Manifest.permission.CAMERA};
        for (String p : wanted) if (!has(p)) need.add(p);
        if (Build.VERSION.SDK_INT >= 33 && !has(Manifest.permission.POST_NOTIFICATIONS)) {
            need.add(Manifest.permission.POST_NOTIFICATIONS);
        }
        if (!need.isEmpty()) requestPermissions(need.toArray(new String[0]), REQ_PERMS);
    }

    // Service tabhi chalao jab mic + overlay permission mil chuki ho
    private void startMiyaIfReady() {
        if (!has(Manifest.permission.RECORD_AUDIO)) return;
        if (!Settings.canDrawOverlays(this)) {
            if (!overlayAsked) {
                overlayAsked = true;
                startActivity(new Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION,
                        Uri.parse("package:" + getPackageName())));
            }
            return;
        }
        try {
            startForegroundService(new Intent(this, MiyaService.class));
        } catch (RuntimeException ignored) { }
    }

    private void js(final String script) {
        runOnUiThread(() -> { if (webView != null) webView.evaluateJavascript(script, null); });
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        if (nativeRec != null) { nativeRec.destroy(); nativeRec = null; }
        if (webView != null) { webView.destroy(); webView = null; }
    }

    // ================= JS <-> Android bridge (sirf apne local index.html ke liye) =================
    public class MiyaBridge {

        @JavascriptInterface
        public String setBrightness(int level) {
            if (!Settings.System.canWrite(MainActivity.this)) {
                runOnUiThread(() -> startActivity(new Intent(Settings.ACTION_MANAGE_WRITE_SETTINGS,
                        Uri.parse("package:" + getPackageName()))));
                return "need_permission";
            }
            try {
                int v = Math.max(0, Math.min(255, level));
                Settings.System.putInt(getContentResolver(), Settings.System.SCREEN_BRIGHTNESS_MODE,
                        Settings.System.SCREEN_BRIGHTNESS_MODE_MANUAL);
                Settings.System.putInt(getContentResolver(), Settings.System.SCREEN_BRIGHTNESS, v);
                return "ok";
            } catch (Exception e) {
                return "error";
            }
        }

        @JavascriptInterface
        public String getBatteryInfo() {
            try {
                BatteryManager bm = (BatteryManager) getSystemService(BATTERY_SERVICE);
                JSONObject o = new JSONObject();
                o.put("level", bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY));
                o.put("charging", bm.isCharging());
                return o.toString();
            } catch (Exception e) {
                return "{}";
            }
        }

        @JavascriptInterface
        public String getDeviceInfo() {
            try {
                JSONObject o = new JSONObject();
                o.put("model", Build.MODEL);
                o.put("sdk", Build.VERSION.SDK_INT);
                return o.toString();
            } catch (Exception e) {
                return "{}";
            }
        }

        @JavascriptInterface
        public void openApp(String packageName) {
            try {
                Intent i = getPackageManager().getLaunchIntentForPackage(packageName);
                if (i != null) startActivity(i);
            } catch (Exception ignored) { }
        }

        // window.open WebView me nahi chalta — https link system browser me kholo
        @JavascriptInterface
        public void openUrl(String url) {
            if (url == null || !url.startsWith("https://")) return;
            try {
                Intent i = new Intent(Intent.ACTION_VIEW, Uri.parse(url));
                i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
                startActivity(i);
            } catch (Exception ignored) { }
        }

        // Camera capture: data:image/png;base64,... ko app ke Pictures folder me save karo
        @JavascriptInterface
        public String saveImage(String dataUrl) {
            try {
                if (dataUrl == null || dataUrl.length() > 14_000_000) return "";
                int comma = dataUrl.indexOf(',');
                if (comma < 0) return "";
                byte[] bytes = Base64.decode(dataUrl.substring(comma + 1), Base64.DEFAULT);
                File dir = getExternalFilesDir(Environment.DIRECTORY_PICTURES);
                if (dir == null) dir = getFilesDir();
                File f = new File(dir, "miya-capture-" + System.currentTimeMillis() + ".png");
                try (FileOutputStream fos = new FileOutputStream(f)) { fos.write(bytes); }
                return f.getAbsolutePath();
            } catch (Exception e) {
                return "";
            }
        }

        // WebView me webkitSpeechRecognition nahi chalta — native recognizer se result JS ko bhejo
        @JavascriptInterface
        public void startListening() {
            runOnUiThread(() -> {
                if (!SpeechRecognizer.isRecognitionAvailable(MainActivity.this)) {
                    js("window.onNativeSpeechError&&window.onNativeSpeechError('unavailable')");
                    return;
                }
                if (!has(Manifest.permission.RECORD_AUDIO)) {
                    js("window.onNativeSpeechError&&window.onNativeSpeechError('permission')");
                    return;
                }
                if (nativeRec == null) nativeRec = SpeechRecognizer.createSpeechRecognizer(MainActivity.this);
                nativeRec.setRecognitionListener(new RecognitionListener() {
                    @Override public void onResults(Bundle b) {
                        ArrayList<String> r = b.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION);
                        if (r != null && !r.isEmpty()) {
                            js("window.onNativeSpeech&&window.onNativeSpeech(" + JSONObject.quote(r.get(0)) + ")");
                        } else {
                            js("window.onNativeSpeechError&&window.onNativeSpeechError('nomatch')");
                        }
                    }
                    @Override public void onError(int e) {
                        js("window.onNativeSpeechError&&window.onNativeSpeechError('error" + e + "')");
                    }
                    @Override public void onReadyForSpeech(Bundle b) {}
                    @Override public void onBeginningOfSpeech() {}
                    @Override public void onRmsChanged(float v) {}
                    @Override public void onBufferReceived(byte[] buf) {}
                    @Override public void onEndOfSpeech() {}
                    @Override public void onEvent(int t, Bundle b) {}
                    @Override public void onPartialResults(Bundle b) {}
                });
                Intent i = new Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
                i.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
                i.putExtra(RecognizerIntent.EXTRA_LANGUAGE, "hi-IN");
                nativeRec.startListening(i);
            });
        }
    }
}
JAVA_EOF

package com.miya.app;

import android.Manifest;
import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.content.pm.PackageManager;
import android.provider.Settings;

public class BootReceiver extends BroadcastReceiver {
    @Override
    public void onReceive(Context context, Intent intent) {
        if (!Intent.ACTION_BOOT_COMPLETED.equals(intent.getAction())) return;
        // Permissions mili hon tabhi service chalao, warna Android 14 par crash hota hai
        boolean mic = context.checkSelfPermission(Manifest.permission.RECORD_AUDIO)
                == PackageManager.PERMISSION_GRANTED;
        if (!mic || !Settings.canDrawOverlays(context)) return;
        try {
            context.startForegroundService(new Intent(context, MiyaService.class));
        } catch (RuntimeException ignored) { }
    }
}
