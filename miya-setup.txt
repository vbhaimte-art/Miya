package com.miya.app;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.app.Service;
import android.content.Intent;
import android.graphics.Color;
import android.graphics.PixelFormat;
import android.os.Handler;
import android.os.IBinder;
import android.os.Looper;
import android.provider.Settings;
import android.speech.RecognitionListener;
import android.speech.RecognizerIntent;
import android.speech.SpeechRecognizer;
import android.speech.tts.TextToSpeech;
import android.util.Log;
import android.view.Gravity;
import android.view.WindowManager;
import android.widget.TextView;

import java.io.OutputStream;
import java.net.Socket;
import java.util.ArrayList;
import java.util.Locale;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

public class MiyaService extends Service {

    static final String TERMUX_HOST = "127.0.0.1";
    static final int TERMUX_PORT = 5555;
    
    WindowManager wm;
    TextView overlayView; // WebView ki jagah TextView (stability ke liye)
    WindowManager.LayoutParams lp;
    final Handler ui = new Handler(Looper.getMainLooper());
    SpeechRecognizer rec;
    TextToSpeech tts;
    boolean ttsReady = false;
    
    boolean termuxConnected = false;
    Socket termuxSocket;
    OutputStream termuxOut;
    
    ExecutorService executor = Executors.newSingleThreadExecutor();

    @Override
    public IBinder onBind(Intent i) { return null; }

    @Override
    public int onStartCommand(Intent intent, int flags, int startFg) {
        if (intent != null && "STOP".equals(intent.getAction())) {
            stopSelf();
            return START_NOT_STICKY;
        }
        
        // Permission check
        if (!Settings.canDrawOverlays(this)) {
            Log.d("MIYA", "Overlay permission needed");
            Intent i = new Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION);
            i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            startActivity(i);
            stopSelf();
            return START_NOT_STICKY;
        }

        if (overlayView == null) {
            startFg();          
            buildOverlay();     
            initTts();          
            startListening();   
            tryConnectTermux(); 
        }
        return START_STICKY;
    }

    void startFg() {
        NotificationManager nm = getSystemService(NotificationManager.class);
        if (nm != null) {
            nm.createNotificationChannel(new NotificationChannel("miya", "Miya AI", NotificationManager.IMPORTANCE_LOW));
        }
        
        Intent stop = new Intent(this, MiyaService.class).setAction("STOP");
        PendingIntent pi = PendingIntent.getService(this, 0, stop, PendingIntent.FLAG_IMMUTABLE);
        
        Notification n = new Notification.Builder(this, "miya")
                .setSmallIcon(android.R.drawable.ic_btn_speak_now)
                .setContentTitle("Miya Active")
                .setContentText("Voice Commands ON")
                .addAction(0, "Stop", pi) 
                .build();
                
        startForeground(1, n);
    }

    void buildOverlay() {
        wm = (WindowManager) getSystemService(WINDOW_SERVICE);
        
        // Android ke naye versions ke mutabiq LayoutParams
        int layoutFlag = WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY;

        lp = new WindowManager.LayoutParams(
                WindowManager.LayoutParams.WRAP_CONTENT,
                WindowManager.LayoutParams.WRAP_CONTENT, 
                layoutFlag,
                WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE | WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL,
                PixelFormat.TRANSLUCENT
        );
        lp.gravity = Gravity.TOP | Gravity.CENTER_HORIZONTAL;
        lp.y = 100;

        // Floating TextView setup
        overlayView = new TextView(this);
        overlayView.setBackgroundColor(Color.argb(180, 0, 0, 0));
        overlayView.setTextColor(Color.GREEN);
        overlayView.setTextSize(14);
        overlayView.setText("Miya Ready...");
        overlayView.setPadding(20, 10, 20, 10);

        try {
            wm.addView(overlayView, lp);
        } catch (Exception e) {
            Log.e("MIYA", "Overlay add failed: " + e.getMessage());
        }
    }

    void initTts() {
        tts = new TextToSpeech(this, status -> {
            if (status == TextToSpeech.SUCCESS) {
                tts.setLanguage(new Locale("hi", "IN")); // Hindi/English ke liye
                ttsReady = true;
            }
        });
    }

    void startListening() {
        ui.post(() -> {
            try {
                if (SpeechRecognizer.isRecognitionAvailable(this)) {
                    rec = SpeechRecognizer.createSpeechRecognizer(this);
                    rec.setRecognitionListener(new RecognitionListener() {
                        @Override public void onReadyForSpeech(Bundle params) {}
                        @Override public void onBeginningOfSpeech() {}
                        @Override public void onRmsChanged(float rmsdB) {}
                        @Override public void onBufferReceived(byte[] buffer) {}
                        @Override public void onEndOfSpeech() {}
                        
                        @Override
                        public void onError(int error) {
                            // Dubara listening start karne ke liye
                            restartListening();
                        }

                        @Override
                        public void onResults(Bundle results) {
                            ArrayList<String> matches = results.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION);
                            if (matches != null && !matches.isEmpty()) {
                                String spokenText = matches.get(0);
                                updateOverlayText("Said: " + spokenText);
                                sendToTermux(spokenText);
                            }
                            restartListening();
                        }

                        @Override public void onPartialResults(Bundle partialResults) {}
                        @Override public void onEvent(int eventType, Bundle params) {}
                    });

                    Intent intent = new Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
                    intent.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
                    intent.putExtra(RecognizerIntent.EXTRA_LANGUAGE, "hi-IN");
                    rec.startListening(intent);
                }
            } catch (Exception e) {
                Log.e("MIYA", "Speech error: " + e.getMessage());
            }
        });
    }

    void restartListening() {
        ui.postDelayed(() -> {
            try {
                if (rec != null) {
                    Intent intent = new Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
                    intent.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
                    rec.startListening(intent);
                }
            } catch (Exception e) {
                Log.e("MIYA", "Restart error: "+ e.getMessage());
            }
        }, 1000);
    }

    void tryConnectTermux() {
        executor.execute(() -> {
            try {
                termuxSocket = new Socket(TERMUX_HOST, TERMUX_PORT);
                termuxOut = termuxSocket.getOutputStream();
                termuxConnected = true;
                Log.d("MIYA", "Connected to Termux server!");
            } catch (Exception e) {
                termuxConnected = false;
                Log.e("MIYA", "Termux connection failed: " + e.getMessage());
            }
        });
    }

    void sendToTermux(String text) {
        executor.execute(() -> {
            try {
                if (termuxConnected && termuxOut != null) {
                    termuxOut.write((text + "\n").getBytes());
                    termuxOut.flush();
                }
            } catch (Exception e) {
                termuxConnected = false;
                Log.e("MIYA", "Send failed, reconnecting...");
                tryConnectTermux();
            }
        });
    }

    void updateOverlayText(String text) {
        ui.post(() -> {
            if (overlayView != null) {
                overlayView.setText(text);
            }
        });
    }

    @Override
    public void onDestroy() {
        super.onDestroy();
        if (overlayView != null && wm != null) {
            wm.removeView(overlayView);
        }
        if (rec != null) {
            rec.destroy();
        }
        if (tts != null) {
            tts.stop();
            tts.shutdown();
        }
        try {
            if (termuxSocket != null) termuxSocket.close();
        } catch (Exception ignored) {}
    }
}
