import android.app.Service;

public class MiyaService extends Service {
    static final String WORKER = "https://young-sky-01cdmy-assistant-api.vbhaimte.workers.dev";
    static final String APP_KEY = "PLACEHOLDER";
    static final String PERSONA = "तुम \"मिया\" हो, मेरी निजी असिस्टेंट और दोस्त। मुझसे जिस भाषा में बात करूँ (हिंदी या English), उसी भाषा में जवाब दो। "
            + "जवाब छोटे रखो, 1-3 वाक्य, क्योंकि तुम बोलकर जवाब देती हो। इमोजी और markdown मत इस्तेमाल करो। "
            + "अपनी सीधी और ईमानदार राय दो; मेरी हर बात से सहमत मत होना, ग़लत लगे तो सलीक़े से टोको।";
    static final int RATE = 16000;
    static final int SECONDS = 5;

    WindowManager wm;
    WebView web;
    WindowManager.LayoutParams lp;
    final Handler ui = new Handler(Looper.getMainLooper());
    TextToSpeech tts;
    boolean ttsReady = false, speaking = false, busy = false;
    final List<JSONObject> history = new ArrayList<>();
    final ExecutorService net = Executors.newSingleThreadExecutor();
    float downX, downY;
    int startX, startY;
    boolean moved;

    @Override
    public IBinder onBind(Intent i) {
        return null;
    }

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        if (intent != null && "STOP".equals(intent.getAction())) {
            stopSelf();
            return START_NOT_STICKY;
        }
        if (web == null) {
            startFg();
            buildOverlay();
            initTts();
        }
        return START_STICKY;
    }

    void startFg() {
        NotificationManager nm = getSystemService(NotificationManager.class);
        nm.createNotificationChannel(new NotificationChannel("miya", "मिया", NotificationManager.IMPORTANCE_LOW));
        Intent stop = new Intent(this, MiyaService.class).setAction("STOP");
        PendingIntent pi = PendingIntent.getService(this, 1, stop, PendingIntent.FLAG_IMMUTABLE | PendingIntent.FLAG_UPDATE_CURRENT);
        Notification n = new Notification.Builder(this, "miya")
                .setSmallIcon(android.R.drawable.ic_btn_speak_now)
                .setContentTitle("मिया चल रही है")
                .setContentText("बंद करने के लिए नीचे का बटन दबाएँ")
                .addAction(new Notification.Action.Builder(
                        Icon.createWithResource(this, android.R.drawable.ic_menu_close_clear_cancel), "बंद करें", pi).build())
                .build();
        if (Build.VERSION.SDK_INT >= 30) {
            try {
                startForeground(1, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_MICROPHONE);
            } catch (Exception e) {
                startForeground(1, n);
            }
        } else {
            startForeground(1, n);
        }
    }

    void buildOverlay() {
        final float d = getResources().getDisplayMetrics().density;
        web = new WebView(this);
        web.setBackgroundColor(Color.TRANSPARENT);
        WebSettings s = web.getSettings();
        s.setJavaScriptEnabled(true);
        s.setDomStorageEnabled(true);
        s.setMediaPlaybackRequiresUserGesture(false);
        final WebViewAssetLoader loader = new WebViewAssetLoader.Builder()
                .addPathHandler("/assets/", new WebViewAssetLoader.AssetsPathHandler(this))
                .build();
        web.setWebViewClient(new WebViewClient() {
            @Override
            public WebResourceResponse shouldInterceptRequest(WebView v, WebResourceRequest r) {
                return loader.shouldInterceptRequest(r.getUrl());
            }
        });
        web.loadUrl("https://appassets.androidplatform.net/assets/avatar.html");

        lp = new WindowManager.LayoutParams(
                (int) (230 * d), (int) (320 * d),
                WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
                WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE
                        | WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL
                        | WindowManager.LayoutParams.FLAG_HARDWARE_ACCELERATED,
                PixelFormat.TRANSLUCENT);
        lp.gravity = Gravity.TOP | Gravity.START;
        lp.x = 40;
        lp.y = 400;

        web.setOnTouchListener((v, e) -> {
            switch (e.getActionMasked()) {
                case MotionEvent.ACTION_DOWN:
                    startX = lp.x;
                    startY = lp.y;
                    downX = e.getRawX();
                    downY = e.getRawY();
                    moved = false;
                    break;
                case MotionEvent.ACTION_MOVE:
                    float dx = e.getRawX() - downX, dy = e.getRawY() - downY;
                    if (Math.abs(dx) > 10 * d || Math.abs(dy) > 10 * d) moved = true;
                    if (moved) {
                        lp.x = startX + (int) dx;
                        lp.y = startY + (int) dy;
                        wm.updateViewLayout(web, lp);
                    }
                    break;
                case MotionEvent.ACTION_UP:
                    if (!moved) onTap();
                    break;
            }
            return true;
        });
        wm = (WindowManager) getSystemService(WINDOW_SERVICE);
        wm.addView(web, lp);
    }

    void initTts() {
        tts = new TextToSpeech(this, st -> {
            if (st == TextToSpeech.SUCCESS) {
                tts.setLanguage(new Locale("hi", "IN"));
                tts.setOnUtteranceProgressListener(new UtteranceProgressListener() {
                    @Override
                    public void onStart(String id) {
                        ui.post(() -> { speaking = true; js("setSpeaking(true)"); });
                    }

                    @Override
                    public void onDone(String id) {
                        ui.post(() -> { speaking = false; js("setSpeaking(false)"); });
                    }

                    @Override
                    public void onError(String id) {
                        ui.post(() -> { speaking = false; js("setSpeaking(false)"); });
                    }
                });
                ttsReady = true;
            }
        });
    }

    void js(final String code) {
        ui.post(() -> {
            if (web != null) web.evaluateJavascript(code, null);
        });
    }

    void onTap() {
        if (speaking) {
            if (tts != null) tts.stop();
            speaking = false;
            js("setSpeaking(false)");
            return;
        }
        if (busy) return;
        if (checkSelfPermission(android.Manifest.permission.RECORD_AUDIO)
                != android.content.pm.PackageManager.PERMISSION_GRANTED) {
            js("setStatus('माइक की अनुमति नहीं है। ऐप खोलकर अनुमति दें')");
            return;
        }
        busy = true;
        js("setStatus('सुन रही हूँ… " + SECONDS + " सेकंड बोलिए')");
        net.execute(() -> {
            String reply;
            try {
                byte[] pcm = record();
                ui.post(() -> js("setStatus('सोच रही हूँ…')"));
                reply = callGemini(pcm);
            } catch (Exception e) {
                reply = "ERR:कुछ गड़बड़ हुई: " + e.getMessage();
            }
            final String r = reply;
            ui.post(() -> onReply(r));
        });
    }

    byte[] record() throws Exception {
        int min = AudioRecord.getMinBufferSize(RATE, AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT);
        AudioRecord ar = new AudioRecord(MediaRecorder.AudioSource.MIC, RATE,
                AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT, Math.max(min, RATE * 2));
        if (ar.getState() != AudioRecord.STATE_INITIALIZED) {
            ar.release();
            throw new Exception("माइक शुरू नहीं हो पाया");
        }
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        byte[] buf = new byte[RATE / 5 * 2];
        ar.startRecording();
        long end = System.currentTimeMillis() + SECONDS * 1000L;
        while (System.currentTimeMillis() < end) {
            int n = ar.read(buf, 0, buf.length);
            if (n > 0) out.write(buf, 0, n);
        }
        ar.stop();
        ar.release();
        return out.toByteArray();
    }

    static byte[] wav(byte[] pcm) {
        ByteBuffer b = ByteBuffer.allocate(44 + pcm.length).order(ByteOrder.LITTLE_ENDIAN);
        b.put("RIFF".getBytes());
        b.putInt(36 + pcm.length);
        b.put("WAVE".getBytes());
        b.put("fmt ".getBytes());
        b.putInt(16);
        b.putShort((short) 1);
        b.putShort((short) 1);
        b.putInt(RATE);
        b.putInt(RATE * 2);
        b.putShort((short) 2);
        b.putShort((short) 16);
        b.put("data".getBytes());
        b.putInt(pcm.length);
        b.put(pcm);
        return b.array();
    }

    String callGemini(byte[] pcm) throws Exception {
        JSONObject body = new JSONObject();
        body.put("systemInstruction", new JSONObject().put("parts",
                new JSONArray().put(new JSONObject().put("text", PERSONA))));
        JSONArray contents = new JSONArray();
        synchronized (history) {
            int from = Math.max(0, history.size() - 20);
            for (int i = from; i < history.size(); i++) contents.put(history.get(i));
        }
        JSONObject audio = new JSONObject().put("inlineData", new JSONObject()
                .put("mimeType", "audio/wav")
                .put("data", Base64.encodeToString(wav(pcm), Base64.NO_WRAP)));
        JSONObject prompt = new JSONObject().put("text",
                "यह मेरी आवाज़ है। मेरी बात हिंदी या English में हो सकती है। जिस भाषा में मैंने बोला, उसी में छोटा जवाब दो।");
        contents.put(new JSONObject().put("role", "user").put("parts", new JSONArray().put(audio).put(prompt)));
        body.put("contents", contents);

        HttpURLConnection c = (HttpURLConnection) new URL(WORKER).openConnection();
        c.setRequestMethod("POST");
        c.setRequestProperty("Content-Type", "application/json");
        c.setRequestProperty("x-miya-key", APP_KEY);
        c.setConnectTimeout(15000);
        c.setReadTimeout(60000);
        c.setDoOutput(true);
        try (OutputStream o = c.getOutputStream()) {
            o.write(body.toString().getBytes("UTF-8"));
        }
        InputStream in = c.getResponseCode() < 400 ? c.getInputStream() : c.getErrorStream();
        ByteArrayOutputStream bo = new ByteArrayOutputStream();
        byte[] buf = new byte[4096];
        int n;
        while ((n = in.read(buf)) > 0) bo.write(buf, 0, n);
        JSONObject j = new JSONObject(bo.toString("UTF-8"));
        JSONArray cand = j.optJSONArray("candidates");
        if (cand == null || cand.length() == 0) {
            JSONObject er = j.optJSONObject("error");
            return "ERR:जवाब नहीं मिला। " + (er != null ? er.optString("message") : "");
        }
        JSONArray parts = cand.getJSONObject(0).getJSONObject("content").getJSONArray("parts");
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < parts.length(); i++) sb.append(parts.getJSONObject(i).optString("text"));
        return sb.toString();
    }

    void onReply(String r) {
        busy = false;
        boolean err = r.startsWith("ERR:");
        String shown = err ? r.substring(4) : r;
        if (!err) {
            synchronized (history) {
                history.add(msgOf("user", "(आवाज़ संदेश)"));
                history.add(msgOf("model", r));
            }
        }
        js("setText(" + JSONObject.quote(shown) + ")");
        if (!err && ttsReady) {
            boolean hi = shown.matches("(?s).*[\\u0900-\\u097F].*");
            tts.setLanguage(hi ? new Locale("hi", "IN") : Locale.US);
            tts.speak(shown, TextToSpeech.QUEUE_FLUSH, null, "miya");
        }
    }

    static JSONObject msgOf(String role, String text) {
        try {
            return new JSONObject().put("role", role)
                    .put("parts", new JSONArray().put(new JSONObject().put("text", text)));
        } catch (Exception e) {
            return new JSONObject();
        }
    }

    @Override
    public void onDestroy() {
        try {
            if (web != null) {
                wm.removeView(web);
                web.destroy();
                web = null;
            }
        } catch (Exception e) {
        }
        if (tts != null) tts.shutdown();
        net.shutdownNow();
        super.onDestroy();
    }
}
