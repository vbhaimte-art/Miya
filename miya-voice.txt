function scanNetwork() {
    const logEl = $('net-log');
    logEl.innerHTML = '';
    show('net-panel');
    let count = 0;
    const ips = ['192.168.1.1', '10.0.0.5', '172.16.0.1', '192.168.4.22', '10.10.10.10'];
    
    const timer = setInterval(() => {
        if(count >= ips.length) {
            clearInterval(timer);
            logEl.innerHTML += "\n[+] Scan Complete.";
            return;
        }
        const ip = ips[count];
        const status = Math.random() > 0.3 ? "OPEN" : "CLOSED";
        logEl.innerHTML += `\n[*] Scanning ${ip}... Status: ${status}`;
        logEl.scrollTop = logEl.scrollHeight;
        count++;
    }, 500);
    activeIntervals.push(timer);
}

}

// Voice Listening (Simplified)
function startListening() {
    const recognition = new (window.SpeechRecognition || window.webkitSpeechRecognition)();
    recognition.lang = 'en-US';
    recognition.onresult = (event) => {
        const transcript = event.results[0][0].transcript.toLowerCase();
        log(`Heard: "${transcript}"`);
        
        // Check commands
        for(const [cmd, func] of Object.entries(commands)) {
            if(transcript.includes(cmd)) {
                func();
                return;
            }
        }
        
        // Check custom tools
        const foundTool = tools.find(t => transcript.includes(t.name.toLowerCase()));
        if(foundTool) {
            runTool(foundTool);
            return;
        }
        
        // Check local brain
        if(!localBrain(transcript)) {
            log("Command not recognized. Try 'Brain' to ask me.");
        }
    };
    recognition.start();
}
