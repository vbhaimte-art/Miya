    void buildOverlay() {
        final float d = getResources().getDisplayMetrics().density;
        web = new WebView(this);
        web.setBackgroundColor(Color.TRANSPARENT);
        WebSettings s = web.getSettings();
        s.setJavaScriptEnabled(true);
        s.setDomStorageEnabled(true);
        s.setMediaPlaybackRequiresUserGesture(false);
        final WebViewAssetLoader loader = new WebViewAssetLoader.Builder()
                .addPathHandler("/assets/", new WebViewAssetLoader.AssetsPathHandler(this))
                .build();
        web.setWebViewClient(new WebViewClient() {
            @Override
            public WebResourceResponse shouldInterceptRequest(WebView v, WebResourceRequest r) {
                return loader.shouldInterceptRequest(r.getUrl());
            }
        });
        web.loadUrl("https://appassets.androidplatform.net/assets/avatar.html");

        lp = new WindowManager.LayoutParams(
                (int) (230 * d), (int) (320 * d),
                WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
                WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE
                        | WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL
                        | WindowManager.LayoutParams.FLAG_HARDWARE_ACCELERATED,
                PixelFormat.TRANSLUCENT);
        lp.gravity = Gravity.TOP | Gravity.START;
        lp.x = 40;
        lp.y = 400;

        web.setOnTouchListener((v, e) -> {
            switch (e.getActionMasked()) {
                case MotionEvent.ACTION_DOWN:
                    startX = lp.x;
                    startY = lp.y;
                    downX = e.getRawX();
                    downY = e.getRawY();
                    moved = false;
                    break;
                case MotionEvent.ACTION_MOVE:
                    float dx = e.getRawX() - downX, dy = e.getRawY() - downY;
                    if (Math.abs(dx) > 10 * d || Math.abs(dy) > 10 * d) moved = true;
                    if (moved) {
                        lp.x = startX + (int) dx;
                        lp.y = startY + (int) dy;
                        wm.updateViewLayout(web, lp);
                    }
                    break;
                case MotionEvent.ACTION_UP:
                    if (!moved) onTap();
                    break;
            }
            return true;
        });
        wm = (WindowManager) getSystemService(WINDOW_SERVICE);
        wm.addView(web, lp);
    }

    void initTts() {
        tts = new TextToSpeech(this, st -> {
            if (st == TextToSpeech.SUCCESS) {
                tts.setLanguage(new Locale("hi", "IN"));
                tts.setOnUtteranceProgressListener(new UtteranceProgressListener() {
                    @Override
                    public void onStart(String id) {
                        ui.post(() -> { speaking = true; js("setSpeaking(true)"); });
                    }

                    @Override
                    public void onDone(String id) {
                        ui.post(() -> { speaking = false; js("setSpeaking(false)"); });
                    }

                    @Override
                    public void onError(String id) {
                        ui.post(() -> { speaking = false; js("setSpeaking(false)"); });
                    }
                });
                ttsReady = true;
            }
        });
    }

    void js(final String code) {

    }

    void onTap() {
        markActive();
        if (speaking) {
            if (tts != null) tts.stop();
            speaking = false;
            js("setSpeaking(false)");
            return;
        }
        if (busy) return;
        if (listening) {
            if (!passive) return;
            try { rec.cancel(); } catch (Exception e) { }
            listening = false;
        }
        passive = false;
        if (checkSelfPermission(android.Manifest.permission.RECORD_AUDIO)
                != PackageManager.PERMISSION_GRANTED) {
            js("setStatus('माइक की अनुमति नहीं है। ऐप खोलकर अनुमति दें')");
            return;
        }
        if (!SpeechRecognizer.isRecognitionAvailable(this)) {
            js("setStatus('इस फ़ोन में आवाज़ पहचान उपलब्ध नहीं है')");
            return;
        }
        listen();
    }

    void listen() {
        if (rec == null) {
            rec = SpeechRecognizer.createSpeechRecognizer(this);
            rec.setRecognitionListener(new RecognitionListener() {
                public void onReadyForSpeech(Bundle p) { }
                public void onBeginningOfSpeech() { }
                public void onRmsChanged(float v) { }
                public void onBufferReceived(byte[] b) { }
                public void onEndOfSpeech() { }
                public void onPartialResults(Bundle b) { }
                public void onEvent(int t, Bundle b) { }

                public void onError(int code) {
                    listening = false;
                    if ((code == 11 || code == 12 || code == 13) && preferOffline) {
                        preferOffline = false;
                        try { rec.destroy(); } catch (Exception e) { }
                        rec = null;
                        ui.postDelayed(() -> listen(), 300);
                        return;
                    }
                    if (passive) {
                        if (code == 8 || code == 5 || code == 11) {
                            try { rec.destroy(); } catch (Exception e) { }
                            rec = null;
                        }
                        return;
                    }
                    if (code == 8 && rec != null) rec.cancel();
                    String why;
                    switch (code) {
                        case 1: why = "नेटवर्क टाइमआउट"; break;
                        case 2: why = "नेटवर्क की दिक्कत"; break;
                        case 3: why = "माइक की दिक्कत"; break;
                        case 4: why = "सर्वर की दिक्कत"; break;
                        case 6: why = "कुछ बोला नहीं गया"; break;
                        case 7: why = "समझ नहीं आया"; break;
                        case 8: why = "पहचान व्यस्त है"; break;
                        case 9: why = "माइक की अनुमति नहीं है"; break;
                        case 10: why = "बहुत ज़्यादा अनुरोध"; break;
                        case 11: why = "आवाज़ सेवा बंद हो गई"; break;
                        case 12: case 13: why = "भाषा उपलब्ध नहीं"; break;
                        default: why = "अज्ञात"; break;
                    }
                    js("setStatus(" + JSONObject.quote("त्रुटि " + code + ": " + why + ", फिर दबाइए") + ")");
                }

                public void onResults(Bundle b) {
                    listening = false;
                    ArrayList<String> r = b.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION);
                    if (passive) { onWake(r); return; }
                    if (r == null || r.isEmpty()) {
                        js("setStatus('समझ नहीं आया, फिर दबाइए')");
                        return;
                    }
                    handle(r, true);
                }
            });
        }
        Intent i = new Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
        i.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
        i.putExtra(RecognizerIntent.EXTRA_LANGUAGE, "hi-IN");
        i.putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 5);
        i.putExtra(RecognizerIntent.EXTRA_PREFER_OFFLINE, preferOffline);
        if (!passive) js("setStatus('सुन रही हूँ…')");
        listening = true;
        rec.startListening(i);
    }

    // 0 = डांस कमांड नहीं, 1 = डांस चालू, 2 = डांस बंद
    int danceCmd(String raw) {
        String s = raw.toLowerCase(Locale.ROOT);
        boolean stopW = has(s, "रुक", "बंद", "बन्द", "रोक", "बस कर", "stop", "band");
        boolean danceW = has(s, "डांस", "डान्स", "डैंस", "नाच", "dance");
        if (stopW && (danceW || dancing)) return 2;
        if (danceW) return 1;
        return 0;
    }

    int modelCmd(String raw) {
        String s = raw.toLowerCase(Locale.ROOT);
        if (has(s, THRON_W)) {
            boolean come = has(s, "आजा", "आ जा", "आओ", "आइए", "बुला", "aajao", "aa jao", "ajao", "aao");
            if (!come && has(s, "जाओ", "जाइए", "वापस", "चले जा", "jao", "jaao")) return 2;
            return 3;
        }
        if (has(s, "योर", "यॉर", "यौर", "yor")) return 1;
        if (has(s, "वापस", "पहले वाली", "पुरानी", "मिया बन", "miya ban", "wapas")) return 2;
        return 0;
    }

    boolean hasWake(String s) {
        return has(s, "मिया", "मीया", "miya", "योर", "यॉर", "यौर", "yor") || has(s, THRON_W);
    }

    String nameOf(List<String> alts) {
        for (String a : alts) {
            String s = a.toLowerCase(Locale.ROOT);
            if (has(s, THRON_W)) return "thron";
            if (has(s, "योर", "यॉर", "यौर", "yor")) return "yor";
            if (has(s, "मिया", "मीया", "miya")) return "miya";
        }
        return null;
    }

    void onWake(List<String> alts) {
        if (alts == null || alts.isEmpty()) return;
        boolean hit = false;
        for (String a : alts) {
            if (hasWake(a.toLowerCase(Locale.ROOT))) { hit = true; break; }
        }
        if (!hit) {
            // नाम नहीं लिया, पर अभी बात चल रही है: सिर्फ़ फ़ोन के कमांड चलाओ
            if (System.currentTimeMillis() < cmdUntil) {
                cmdUntil = System.currentTimeMillis() + CMD_WINDOW_MS;
                handle(alts, false);
            }
            return;
        }
        showChar();
        cmdUntil = System.currentTimeMillis() + CMD_WINDOW_MS;
        String rest = alts.get(0).toLowerCase(Locale.ROOT)
                .replaceAll("मिया|मीया|miya|योर|यॉर|यौर|yor|थ्रोन|थ्रॉन|थ्रोण|थोर्न|थॉर्न|ट्रोन|throne|thron|thorn", " ")
                .replaceAll("[\\s,।.!?]+", " ").trim();
        if (rest.length() < 2) {
            String nc = nameOf(alts);
            if (nc != null && !nc.equals(curChar)) {
                js("setModel('" + (nc.equals("miya") ? "model" : nc) + "')");
                switchChar(nc);
            }
            awaitCmd = true;
            js("setStatus('जी?')");
            ui.removeCallbacks(wakeTick);
            ui.postDelayed(wakeTick, 300);
        } else {
            handle(alts, true);
        }
    }

    String curChar = "miya";

    android.content.SharedPreferences store() {
        return getSharedPreferences("miya_store", MODE_PRIVATE);

    String personaNow() {
        String base;
        if (curChar.equals("yor")) {
            base = PERSONA.replace("\"मिया\"", "\"योर\"") + YOR_STYLE;
        } else if (curChar.equals("thron")) {
            base = THRON_PERSONA;
        } else {
            base = PERSONA;
        }
        String mem = store().getString("mem_" + curChar, "");
        if (mem.isEmpty()) return base;
        return base + "\nमेरे बारे में तुम्हें ये बातें याद हैं, इनका ध्यान रखो:\n" + mem;
    }

    void saveHist() {
        try {
            JSONArray a = new JSONArray();
            synchronized (history) {
                int from = Math.max(0, history.size() - 40);
                for (int i = from; i < history.size(); i++) a.put(history.get(i));
            }
            store().edit().putString("hist_" + curChar, a.toString()).apply();
        } catch (Exception e) { }
    }

    void loadHist() {
        synchronized (history) {
            history.clear();
            try {
                String s = store().getString("hist_" + curChar, null);
                if (s != null) {
                    JSONArray a = new JSONArray(s);
                    for (int i = 0; i < a.length(); i++) history.add(a.getJSONObject(i));
                }
            } catch (Exception e) { }
        }
    }

    void switchChar(String c) {
        if (c.equals(curChar)) return;
        saveHist();
        curChar = c;
        loadHist();
        applySize();
    }

    String tokClean(String s, String... stop) {
        List<String> st = Arrays.asList(stop);
        StringBuilder sb = new StringBuilder();
        for (String t : s.split("[\\s,।.!?]+")) {
            if (!t.isEmpty() && !st.contains(t)) sb.append(t).append(' ');
        }
        return sb.toString().trim();
    }

    String memCmd(String s) {
        String key = "mem_" + curChar;
        if (has(s, "सब भूल", "याद मिटा", "याद साफ")) {
            store().edit().remove(key).remove("hist_" + curChar).apply();
            synchronized (history) { history.clear(); }
            return "ठीक है, सब भूल गई।";
        }
        if (has(s, "क्या याद है", "क्या याद रखा", "मेरे बारे में क्या")) {

    int pickHour(String s, int h) {
        if (has(s, "शाम") || has(s, " pm")) {

            if (h == 12) h = 0;
            else if (h >= 6 && h < 12) h += 12;
        }
        return h;
    }

    // allowAsk = false: सिर्फ़ फ़ोन के कमांड, Gemini से बात नहीं (बिना नाम के सुनी बात के लिए)
    void handle(List<String> alts, boolean allowAsk) {
        for (String a : alts) {
            String low = a.toLowerCase(Locale.ROOT);
            if (allowAsk && deepCmd(a, low)) return;
            String tc = thronCmd(low);
            if (tc != null) { if (!tc.isEmpty()) say(tc); return; }
            String mm = memCmd(low);
            if (mm != null) { say(mm); return; }
            String ex = extra(low);
            if (ex != null) { say(ex); return; }
            String r;
            try {
                r = device(low);
            } catch (Exception e) {
                r = "यह काम नहीं हो पाया।";
            }
            if (r != null) {
                boolean bad = r.contains("नहीं") || r.contains("?") || r.contains("कितने");
                if (bad) say(r); else quiet();
                return;
            }
            int mc = modelCmd(a);
            if (mc != 0) {
                showChar();
                js("setModel('" + (mc == 1 ? "yor" : mc == 3 ? "thron" : "model") + "')");
                switchChar(mc == 1 ? "yor" : mc == 3 ? "thron" : "miya");
                say(mc == 1 ? "ठीक है, अब मैं योर हूँ।" : mc == 3 ? "जी बॉस, बताइए।" : "ठीक है, मैं वापस मिया बन गई।");
                return;
            }
            int c = danceCmd(a);
            if (c == 1) {
                showChar();
                setDance(true);
                quiet();
                return;
            }
            if (c == 2) {
                setDance(false);
                quiet();
                return;
            }
        }
        if (allowAsk) ask(alts.get(0));
    }

    String device(String s) {

        if (timerW || (alarmW && unitW && has(s, "बाद", "after", "later"))) return timerReply(s);
        if (alarmW) {
            if (has(s, "बंद", "बन्द", "हटा", "cancel", "delete", "off", "दिखा", "show")) {
                try {
                    startActivity(new Intent(AlarmClock.ACTION_SHOW_ALARMS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK));
                    return "अलार्म की सूची खोल दी है, वहाँ से बंद कर लीजिए।";

    }

    // एक-दो जैसे हिंदी अंक-शब्द को संख्या में बदलता है, पर सिर्फ़ तब जब उसके बाद बजे/मिनट/सेकंड/घंटा आए
    String normNum(String s) {
        String[] t = s.split("\\s+");
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < t.length; i++) {
            String w = t[i];
            String nx = i + 1 < t.length ? t[i + 1] : "";
            boolean unit = nx.startsWith("बज") || nx.startsWith("मिन") || nx.startsWith("सेक")
                    || nx.startsWith("घंट") || nx.startsWith("घण्ट") || nx.startsWith("min")
                    || nx.startsWith("sec") || nx.startsWith("hour");
            if (unit) {
                for (String[] p : NUMW) {
                    if (p[0].equals(w)) { w = p[1]; break; }
                }
            }
            sb.append(w).append(' ');
        }
        return sb.toString();
    }

    // ---------- 1. ऐप खोलना ----------
    String openReply(String s) {
        List<String> stop = Arrays.asList("खोलो", "खोल", "खोलना", "दो", "दे", "करो", "कर", "ओपन", "open",
                "लॉन्च", "launch", "मिया", "miya", "प्लीज़", "प्लीज", "please", "ज़रा", "जरा",
                "एप्लिकेशन", "application", "ऐप", "एप", "app", "को", "जी");
        StringBuilder sb = new StringBuilder();
        for (String t : s.split("\\s+")) {
            if (!t.isEmpty() && !stop.contains(t)) sb.append(t).append(' ');
        }
        String name = sb.toString().trim();
        if (name.isEmpty()) return "कौन सा ऐप खोलूँ?";
        for (String[] p : APPS) {
            if (name.contains(p[0])) { name = p[1]; break; }
        }
        PackageManager pm = getPackageManager();
        Intent li = new Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_LAUNCHER);
        String best = null, bestLabel = "";
        int bestScore = 0;
        for (ResolveInfo ri : pm.queryIntentActivities(li, 0)) {
            String label = ri.loadLabel(pm).toString().toLowerCase(Locale.ROOT);
            int sc = label.equals(name) ? 3 : label.startsWith(name) ? 2 : label.contains(name) ? 1 : 0;
            if (sc > bestScore || (sc == bestScore && sc > 0 && label.length() < bestLabel.length())) {
                bestScore = sc;
                best = ri.activityInfo.packageName;
                bestLabel = label;
            }
        }
        if (best == null) {
            return name.split(" ").length <= 3 ? "मुझे " + name + " नाम का ऐप नहीं मिला।" : null;
        }
        Intent launch = pm.getLaunchIntentForPackage(best);
        if (launch == null) return "वह ऐप खुल नहीं पाया।";
        launch.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
        startActivity(launch);
        return bestLabel + " खोल रही हूँ।";
    }

    // ---------- 2. अलार्म और टाइमर ----------
    String timerReply(String s) {
        try {
            String n = normNum(s);
            Matcher m = Pattern.compile("(\\d+)\\s*(घंट|घण्ट|hour|मिनट|मिनिट|minute|min|सेकंड|सेकेंड|second|sec)").matcher(n);
            int sec = 0;
            while (m.find()) {
                int v = Integer.parseInt(m.group(1));
                String u = m.group(2);
                if (u.startsWith("घ") || u.startsWith("hour")) sec += v * 3600;
                else if (u.startsWith("म") || u.startsWith("min")) sec += v * 60;
                else sec += v;
            }
            if (sec <= 0) return "कितने समय का टाइमर लगाऊँ? जैसे: पाँच मिनट का टाइमर";
            if (sec > 86400) return "इतना लंबा टाइमर नहीं लगा सकती, 24 घंटे तक ही।";
            startActivity(new Intent(AlarmClock.ACTION_SET_TIMER)
                    .putExtra(AlarmClock.EXTRA_LENGTH, sec)
                    .putExtra(AlarmClock.EXTRA_MESSAGE, "मिया")
                    .putExtra(AlarmClock.EXTRA_SKIP_UI, true)
                    .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK));
            int hh = sec / 3600, mm = (sec % 3600) / 60, ss = sec % 60;
            StringBuilder d = new StringBuilder();
            if (hh > 0) d.append(hh).append(" घंटे ");
            if (mm > 0) d.append(mm).append(" मिनट ");
            if (ss > 0) d.append(ss).append(" सेकंड");
            return d.toString().trim() + " का टाइमर लगा दिया।";
        } catch (Exception e) {
            return "टाइमर नहीं लग पाया।";
        }
    }

    String alarmReply(String s) {
        try {
            String n = normNum(s);
            int h = -1, mi = 0;
            Matcher c = Pattern.compile("(\\d{1,2})\\s*[:.]\\s*(\\d{2})").matcher(n);
            if (c.find()) {
                h = Integer.parseInt(c.group(1));
                mi = Integer.parseInt(c.group(2));
            } else {
                Matcher m = Pattern.compile("\\d+").matcher(n);
                if (m.find()) {
                    h = Integer.parseInt(m.group());
                    if (m.find()) {
                        int x = Integer.parseInt(m.group());
                        if (x <= 59) mi = x;
                    }
                }
            }
            if (h < 0 || h > 23 || mi > 59) {
                return "कितने बजे का अलार्म लगाऊँ? जैसे: सुबह छह बजे का अलार्म लगाओ";
            }
            if (has(n, "साढ़े", "साढे", "साड़े")) mi = 30;
            else if (has(n, "सवा")) mi = 15;
            boolean paune = has(n, "पौने");
            boolean pm = has(s, " pm", "p.m");
            if (has(s, "शाम") || pm) {
                if (h < 12) h += 12;
            } else if (has(s, "दोपहर")) {
                if (h >= 1 && h <= 6) h += 12;
            } else if (has(s, "रात")) {
                if (h == 12) h = 0;
                else if (h >= 6 && h < 12) h += 12;
            }
            if (paune) {

    void onBattery(Intent i) {
        int lvl = i.getIntExtra(android.os.BatteryManager.EXTRA_LEVEL, -1);
        int scale = i.getIntExtra(android.os.BatteryManager.EXTRA_SCALE, 100);
        if (lvl < 0 || scale <= 0) return;
        int pct = lvl * 100 / scale;
        int st = i.getIntExtra(android.os.BatteryManager.EXTRA_STATUS, -1);
        boolean chg = st == android.os.BatteryManager.BATTERY_STATUS_CHARGING
                || st == android.os.BatteryManager.BATTERY_STATUS_FULL;
        int temp = i.getIntExtra(android.os.BatteryManager.EXTRA_TEMPERATURE, 0) / 10;
        long now = System.currentTimeMillis();
        batPct = pct;
        batChg = chg;

        if (chg) {
            warnStage = 0;
            dropRefPct = -1;
            if (autoSaved) { wakeOn = true; autoSaved = false; }
            if (pct >= 100 && !fullSaid) {
                fullSaid = true;
                thronAlert("बैटरी full है, चार्जर निकाल दीजिए।", false);
            }
            if (pct < 95) fullSaid = false;
        } else {
            fullSaid = false;
            if (pct <= 20 && wakeOn) { wakeOn = false; autoSaved = true; }
            if (pct > 25) warnStage = 0;
            if (pct <= 5 && warnStage < 3) {
                warnStage = 3;
                thronAlert("बैटरी सिर्फ़ " + pct + " प्रतिशत है, फ़ोन बंद होने वाला है। अभी चार्जर लगाइए।", true);
            } else if (pct <= 10 && warnStage < 2) {
                warnStage = 2;
                thronAlert("बैटरी " + pct + " प्रतिशत रह गई है, चार्जर लगाइए।", true);
            } else if (pct <= 20 && warnStage < 1) {
                warnStage = 1;
                thronAlert("बैटरी " + pct + " प्रतिशत है। बैटरी बचाने के लिए हमेशा सुनना बंद कर दिया है।", false);
            }
            if (dropRefPct < 0) {
                dropRefPct = pct;
                dropRefT = now;
            } else if (now - dropRefT >= 15 * 60000L) {
                int drop = dropRefPct - pct;
                if (drop >= 15 && now - lastDrainAt > 3600000L) {
                    lastDrainAt = now;
                    thronAlert("बैटरी 15 मिनट में " + drop + " प्रतिशत गिरी है। कोई ऐप पीछे चल रही हो सकती है, सेटिंग में बैटरी उपयोग देखिए।", false);
                }
                dropRefPct = pct;
                dropRefT = now;
            }
        }
        if (temp >= 42 && now - lastHeatAt > 600000L) {
            lastHeatAt = now;
            thronAlert("फ़ोन गरम है (" + temp + " डिग्री)। भारी ऐप बंद कर दीजिए।", temp >= 45);
        }
    }

    String buildScan() {
        List<String> out = new ArrayList<>();
        try {
            android.app.KeyguardManager km = (android.app.KeyguardManager) getSystemService(KEYGUARD_SERVICE);
            if (km != null && !km.isDeviceSecure()) out.add("फ़ोन पर स्क्रीन लॉक नहीं है");
        } catch (Exception e) { }
        try {
            if (android.provider.Settings.Global.getInt(getContentResolver(),
                    android.provider.Settings.Global.ADB_ENABLED, 0) == 1) out.add("USB डिबगिंग चालू है");
        } catch (Exception e) { }
        try {
            PackageManager pm = getPackageManager();
            int mic = 0, side = 0;
            List<String> names = new ArrayList<>();
            for (android.content.pm.PackageInfo p : pm.getInstalledPackages(PackageManager.GET_PERMISSIONS)) {
                android.content.pm.ApplicationInfo ai = p.applicationInfo;
                if (ai == null || (ai.flags & android.content.pm.ApplicationInfo.FLAG_SYSTEM) != 0) continue;
                if (p.packageName.equals(getPackageName())) continue;
                if (p.requestedPermissions != null && p.requestedPermissionsFlags != null) {
                    for (int i = 0; i < p.requestedPermissions.length && i < p.requestedPermissionsFlags.length; i++) {
                        if (android.Manifest.permission.RECORD_AUDIO.equals(p.requestedPermissions[i])
                                && (p.requestedPermissionsFlags[i] & android.content.pm.PackageInfo.REQUESTED_PERMISSION_GRANTED) != 0) {
                            mic++;
                            break;
                        }
                    }
                }
                if (installerOf(p.packageName) == null) {
                    side++;
                    if (names.size() < 3) names.add(String.valueOf(pm.getApplicationLabel(ai)));
                }
            }
            if (mic > 0) out.add(mic + " ऐप्स के पास माइक की अनुमति है");
            if (side > 0) out.add("प्ले स्टोर के बाहर से आई " + side + " ऐप हैं, जैसे " + android.text.TextUtils.join(", ", names));
        } catch (Exception e) { }
        if (out.isEmpty()) return "बॉस, स्कैन पूरा, सब ठीक दिख रहा है।";
        return "बॉस, स्कैन पूरा। " + android.text.TextUtils.join("। ", out) + "।";
    }

                }

            f.addDataScheme("package");
            registerReceiver(pkgRx, f);
        } catch (Exception e) { }
    }

    String postWorker(JSONObject body) throws Exception {
        HttpURLConnection c = (HttpURLConnection) new URL(WORKER).openConnection();
        c.setRequestMethod("POST");
        c.setRequestProperty("Content-Type", "application/json");
        c.setRequestProperty("x-miya-key", APP_KEY);
        c.setConnectTimeout(15000);
        c.setReadTimeout(90000);
        c.setDoOutput(true);
        try (OutputStream o = c.getOutputStream()) {
            o.write(body.toString().getBytes("UTF-8"));
        }
        InputStream in = c.getResponseCode() < 400 ? c.getInputStream() : c.getErrorStream();
        ByteArrayOutputStream bo = new ByteArrayOutputStream();
        byte[] buf = new byte[4096];
        int n;
        while ((n = in.read(buf)) > 0) bo.write(buf, 0, n);
        JSONObject j = new JSONObject(bo.toString("UTF-8"));
        JSONArray cand = j.optJSONArray("candidates");
        if (cand == null || cand.length() == 0) {
            JSONObject er = j.optJSONObject("error");
            return "ERR:जवाब नहीं मिला। " + (er != null ? er.optString("message") : "");
        }
        JSONArray parts = cand.getJSONObject(0).getJSONObject("content").getJSONArray("parts");
        StringBuilder sb = new StringBuilder();

        return sb.toString();
    }

    byte[] readFile(java.io.File f) throws Exception {
        try (java.io.FileInputStream in = new java.io.FileInputStream(f)) {
            ByteArrayOutputStream bo = new ByteArrayOutputStream();
            byte[] buf = new byte[8192];
            int n;
            while ((n = in.read(buf)) > 0) bo.write(buf, 0, n);
            return bo.toByteArray();
        }
    }

    String callShare(java.io.File img, java.io.File txt) throws Exception {
        String ask = "ये मेरे फ़ोन से भेजी गई चीज़ है। अगर इसमें कोई build या कोड की error है, तो बताओ कि गलती कहाँ है और कैसे ठीक करें, कदम-दर-कदम। "

        JSONArray parts = new JSONArray().put(new JSONObject().put("text", ask));
        if (img.exists()) {
            String b64 = android.util.Base64.encodeToString(readFile(img), android.util.Base64.NO_WRAP);
            parts.put(new JSONObject().put("inline_data",
                    new JSONObject().put("mime_type", "image/jpeg").put("data", b64)));
        } else if (txt.exists()) {
            String t = new String(readFile(txt), "UTF-8");
            if (t.length() > 20000) t = t.substring(t.length() - 20000);
            parts.put(new JSONObject().put("text", t));
        } else {
            return "ERR:कुछ मिला नहीं, फिर से शेयर कीजिए।";
        }
        JSONObject body = new JSONObject();
        String sys = personaNow()
                + "\nअब विस्तार से जवाब दो। सबसे ऊपर पहली पंक्ति में 1-2 वाक्य का सार बोलचाल की भाषा में लिखो, "
                + "फिर एक खाली पंक्ति छोड़कर पूरा जवाब क्रमवार दो।";
        body.put("systemInstruction", new JSONObject().put("parts", new JSONArray().put(new JSONObject().put("text", sys))));
        body.put("contents", new JSONArray().put(new JSONObject().put("role", "user").put("parts", parts)));
        body.put("generationConfig", new JSONObject().put("maxOutputTokens", 4096));

    void longNotify(String text) {
        try {
            if (text.length() > 4000) text = text.substring(0, 4000);
            NotificationManager nm = getSystemService(NotificationManager.class);
            nm.createNotificationChannel(new NotificationChannel("thron", "थ्रोन", NotificationManager.IMPORTANCE_DEFAULT));
            Notification n = new Notification.Builder(this, "thron")
                    .setSmallIcon(android.R.drawable.ic_dialog_info)
                    .setContentTitle("थ्रोन का जवाब")
                    .setContentText(text.length() > 80 ? text.substring(0, 80) : text)
                    .setStyle(new Notification.BigTextStyle().bigText(text))
                    .setAutoCancel(true)
                    .build();
            nm.notify(3, n);
        } catch (Exception e) { }
    }

    String saveLast() {
        if (lastLong.isEmpty()) return "अभी सेव करने के लिए कुछ नहीं है।";
        String name = "thron_" + new java.text.SimpleDateFormat("yyyyMMdd_HHmmss", Locale.US).format(new java.util.Date()) + ".txt";
        try {
            byte[] data = lastLong.getBytes("UTF-8");
            if (Build.VERSION.SDK_INT >= 29) {
                android.content.ContentValues v = new android.content.ContentValues();
                v.put(android.provider.MediaStore.MediaColumns.DISPLAY_NAME, name);
                v.put(android.provider.MediaStore.MediaColumns.MIME_TYPE, "text/plain");
                v.put(android.provider.MediaStore.MediaColumns.RELATIVE_PATH, "Download");
                android.net.Uri u = getContentResolver().insert(android.provider.MediaStore.Downloads.EXTERNAL_CONTENT_URI, v);
                if (u == null) return "फाइल नहीं बन पाई।";
                try (OutputStream o = getContentResolver().openOutputStream(u)) {
                    o.write(data);
                }
                return "फाइल Downloads में सेव कर दी, नाम " + name;
            }
            java.io.File dir = getExternalFilesDir(null);
            if (dir == null) return "फाइल नहीं बन पाई।";
            try (OutputStream o = new java.io.FileOutputStream(new java.io.File(dir, name))) {
                o.write(data);
            }
            return "फाइल ऐप के फोल्डर में सेव कर दी, नाम " + name;
        } catch (Exception e) {
            return "फाइल नहीं बन पाई।";
        }
    }

    String shareLast() {
        if (lastLong.isEmpty()) return "अभी शेयर करने के लिए कुछ नहीं है।";

            return "शेयर नहीं खुल पाया।";
        }
    }

    String callGemini(String userText) throws Exception {
        JSONObject body = new JSONObject();
        body.put("systemInstruction", new JSONObject().put("parts",
                new JSONArray().put(new JSONObject().put("text", personaNow()))));
        JSONArray contents = new JSONArray();
        synchronized (history) {
            int from = Math.max(0, history.size() - 20);
            while (from < history.size() && !"user".equals(history.get(from).optString("role"))) from++;
            for (int i = from; i < history.size(); i++) contents.put(history.get(i));
        }
        contents.put(msgOf("user", userText));
        body.put("contents", contents);

        HttpURLConnection c = (HttpURLConnection) new URL(WORKER).openConnection();
        c.setRequestMethod("POST");
        c.setRequestProperty("Content-Type", "application/json");
        c.setRequestProperty("x-miya-key", APP_KEY);
        c.setConnectTimeout(15000);
        c.setReadTimeout(60000);
        c.setDoOutput(true);
        try (OutputStream o = c.getOutputStream()) {
            o.write(body.toString().getBytes("UTF-8"));
        }
        InputStream in = c.getResponseCode() < 400 ? c.getInputStream() : c.getErrorStream();
        ByteArrayOutputStream bo = new ByteArrayOutputStream();
        byte[] buf = new byte[4096];
        int n;
        while ((n = in.read(buf)) > 0) bo.write(buf, 0, n);
        JSONObject j = new JSONObject(bo.toString("UTF-8"));
        JSONArray cand = j.optJSONArray("candidates");
        if (cand == null || cand.length() == 0) {
            JSONObject er = j.optJSONObject("error");
            return "ERR:जवाब नहीं मिला। " + (er != null ? er.optString("message") : "");
        }
        JSONArray parts = cand.getJSONObject(0).getJSONObject("content").getJSONArray("parts");
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < parts.length(); i++) sb.append(parts.getJSONObject(i).optString("text"));
        return sb.toString();
    }

    void onReply(String userText, String r) {
        busy = false;
        showChar();
        cmdUntil = System.currentTimeMillis() + CMD_WINDOW_MS;
        boolean err = r.startsWith("ERR:");
        String shown = err ? r.substring(4) : r;
        if (!err) {
            synchronized (history) {
                history.add(msgOf("user", userText));
                history.add(msgOf("model", r));
                while (history.size() > 40) history.remove(0);
                saveHist();
            }
        }
        js("setText(" + JSONObject.quote(shown) + ")");
        if (!err && ttsReady) {
            boolean hi = shown.matches("(?s).*[\\u0900-\\u097F].*");
            tts.setLanguage(hi ? new Locale("hi", "IN") : Locale.US);
            tts.setPitch(curChar.equals("thron") ? 0.8f : 1f);
            tts.speak(shown, TextToSpeech.QUEUE_FLUSH, null, "miya");
        }
    }

    static JSONObject msgOf(String role, String text) {
        try {
            return new JSONObject().put("role", role)
                    .put("parts", new JSONArray().put(new JSONObject().put("text", text)));
        } catch (Exception e) {
            return new JSONObject();
        }
    }
