package com.miya.app;

import android.Manifest;
import android.content.BroadcastReceiver;
import android.content.ClipData;
import android.content.ClipboardManager;
import android.content.Context;
import android.content.Intent;
import android.content.IntentFilter;
import android.content.pm.PackageManager;
import android.net.Uri;
import android.os.BatteryManager;
import android.os.Build;
import android.os.Bundle;
import android.os.SystemClock;
import android.provider.Settings;
import android.util.Log;
import android.view.ViewGroup;
import android.view.WindowManager;
import android.webkit.GeolocationPermissions;
import android.webkit.JavascriptInterface;
import android.webkit.PermissionRequest;
import android.webkit.WebChromeClient;
import android.webkit.WebResourceRequest;
import android.webkit.WebResourceResponse;
import android.webkit.WebSettings;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import android.widget.Toast;

import androidx.activity.OnBackPressedCallback;
import androidx.annotation.NonNull;
import androidx.biometric.BiometricManager;
import androidx.biometric.BiometricPrompt;
import androidx.core.app.ActivityCompat;
import androidx.core.content.ContextCompat;
import androidx.fragment.app.FragmentActivity;
import androidx.webkit.WebViewAssetLoader;

import org.json.JSONObject;

import java.io.ByteArrayInputStream;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Locale;
import java.util.regex.Pattern;

public class MainActivity extends FragmentActivity {

    private static final String TAG = "MiyaMain";
    private static final String HOST = "appassets.androidplatform.net";
    private static final String START_URL = "https://" + HOST + "/assets/index.html";
    private static final int REQ_PERMS = 100;
    private static final int REQ_WEB = 101;
    private static final long SESSION_TIMEOUT_MS = 10 * 60 * 1000L;
    private static final Pattern PKG_PATTERN = Pattern.compile("^[a-zA-Z0-9_.]{1,200}$");
    private static final Pattern TOKEN_PATTERN = Pattern.compile("^[0-9a-fA-F]{32}$");
    private static final Pattern GH_TOKEN_PATTERN =
            Pattern.compile("^(github_pat_[A-Za-z0-9_]{20,}|ghp_[A-Za-z0-9]{20,})$");
    private static final int AUTHENTICATORS =
            BiometricManager.Authenticators.BIOMETRIC_WEAK
                    | BiometricManager.Authenticators.DEVICE_CREDENTIAL;

    private WebView webView;
    private WebViewAssetLoader assetLoader;
    private boolean started = false;
    private boolean voiceStarted = false;
    private PermissionRequest pendingWebReq;

    // MiyaService se wake-word ka text yahan aata hai aur JS ko forward hota hai
    private final BroadcastReceiver wakeReceiver = new BroadcastReceiver() {
        @Override
        public void onReceive(Context context, Intent intent) {
            String text = intent.getStringExtra(MiyaService.EXTRA_TEXT);
            if (text == null) return;

            // Ye do commands sirf app khuli hone par (foreground me) ho sakte hain
            String low = text.toLowerCase(Locale.ROOT).trim();
            if (low.startsWith("termux token")) { saveTermuxTokenFromClipboard(); return; }
            String norm = low.replace(" ", "");
            if (norm.startsWith("githubtoken") || norm.startsWith("gethubtoken")) { saveGithubTokenFromClipboard(); return; }
            if (low.startsWith("agent setup")) { openAccessibilitySettings(); return; }

            if (webView == null) return;
            webView.evaluateJavascript(
                    "window.onMiyaWake && window.onMiyaWake(" + JSONObject.quote(text) + ")", null);
        }
    };
    private boolean wakeReceiverRegistered = false;

    // Session state (elapsedRealtime: user phone ki clock badal ke bypass nahi kar sakta).
    // static: MiyaAgent bhi check karta hai ki fingerprint/PIN se unlock hua hai ya nahi.
    private static volatile boolean unlocked = false;
    private static volatile long unlockedAt = 0;

    /** MiyaAgent (agent / termux commands) isse check karta hai. */
    static boolean sessionActive() {
        if (!unlocked) return false;
        if (SystemClock.elapsedRealtime() - unlockedAt > SESSION_TIMEOUT_MS) {
            unlocked = false;
            return false;
        }
        return true;
    }

    // ------------------------------------------------------------------
    // LIFECYCLE
    // ------------------------------------------------------------------
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);

        // Screenshots / recent-apps preview band
        getWindow().setFlags(WindowManager.LayoutParams.FLAG_SECURE,
                WindowManager.LayoutParams.FLAG_SECURE);

        setupWebView(); // load NAHI hota; pehle security checks

        getOnBackPressedDispatcher().addCallback(this, new OnBackPressedCallback(true) {
            @Override public void handleOnBackPressed() {
                if (webView != null && webView.canGoBack()) webView.goBack(); else finish();
            }
        });

        // Security gate: local checks -> server (Play Integrity) -> UI load
        SecurityModule.runChecksAsync(this, threat -> {
            if (threat != SecurityModule.Threat.NONE) {
                SecurityModule.enforce(this, threat);
            } else {
                verifyServerThenStart();
            }
        });
    }

    @Override
    protected void onStart() {
        super.onStart();
        if (!wakeReceiverRegistered) {
            IntentFilter f = new IntentFilter(MiyaService.ACTION_WAKE);
            // Android 13+/14: exported flag batana zaroori (apni app ke andar hi broadcast)
            if (Build.VERSION.SDK_INT >= 33) {
                registerReceiver(wakeReceiver, f, Context.RECEIVER_NOT_EXPORTED);
            } else {
                registerReceiver(wakeReceiver, f);
            }
            wakeReceiverRegistered = true;
        }
    }

    @Override
    protected void onStop() {
        super.onStop();
        unlocked = false; // app background mein gayi -> dobara auth
        if (wakeReceiverRegistered) {
            try { unregisterReceiver(wakeReceiver); } catch (Exception ignored) {}
            wakeReceiverRegistered = false;
        }
    }

    @Override
    protected void onDestroy() {
        pendingWebReq = null;
        if (webView != null) {
            // destroy se pehle parent se hatao, warna leak/crash ho sakta hai
            if (webView.getParent() instanceof ViewGroup) {
                ((ViewGroup) webView.getParent()).removeView(webView);
            }
            webView.removeJavascriptInterface("Android");
            webView.loadUrl("about:blank");
            webView.destroy();
            webView = null;
        }
        super.onDestroy();
    }

    // ------------------------------------------------------------------
    // SECURITY GATE
    // ------------------------------------------------------------------
    private void verifyServerThenStart() {
        String url = BuildConfig.INTEGRITY_BASE_URL;
        if (BuildConfig.DEBUG || url == null || url.isEmpty()) {
            startApp(); // debug build ya server configure nahi
            return;
        }
        SecurityModule.verifyWithServer(this, url, BuildConfig.CLOUD_PROJECT_NUMBER,
                (ok, reason) -> {
                    if (ok) {
                        startApp();
                    } else {
                        Log.w(TAG, "Server verify failed: " + reason);
                        SecurityModule.showBlockedAndExit(this);
                    }
                });
    }

    private void startApp() {
        // callback activity destroy hone ke baad aaye to webView null hota hai (NPE)
        if (started || isFinishing() || isDestroyed() || webView == null) return;
        started = true;

        requestRuntimePermissions();

        try {
            getAssets().open("index.html").close();
            webView.loadUrl(START_URL);
        } catch (Exception e) {
            Toast.makeText(this, "Error: assets/index.html missing!", Toast.LENGTH_LONG).show();
        }

        // mic-type foreground service tabhi, jab RECORD_AUDIO mil chuki ho.
        // Android 14+ pe bina permission ke service SecurityException se crash hoti hai.
        if (has(Manifest.permission.RECORD_AUDIO)) startVoiceService();
    }

    private void startVoiceService() {
        if (voiceStarted || isFinishing() || isDestroyed()) return;
        if (!has(Manifest.permission.RECORD_AUDIO)) return;
        voiceStarted = true;
        try {
            startService(new Intent(this, MiyaService.class));
        } catch (Exception e) {
            voiceStarted = false;
            Log.w(TAG, "Service start failed", e);
        }
    }

    // ------------------------------------------------------------------
    // WEBVIEW (sirf apni local assets; remote content block)
    // ------------------------------------------------------------------
    private void setupWebView() {
        webView = new WebView(this);
        setContentView(webView);

        assetLoader = new WebViewAssetLoader.Builder()
                .setDomain(HOST)
                .addPathHandler("/assets/", new WebViewAssetLoader.AssetsPathHandler(this))
                .build();

        WebSettings ws = webView.getSettings();
        ws.setJavaScriptEnabled(true);
        ws.setDomStorageEnabled(true);
        ws.setAllowFileAccess(false);
        ws.setAllowContentAccess(false);
        ws.setJavaScriptCanOpenWindowsAutomatically(false);
        ws.setMediaPlaybackRequiresUserGesture(false);
        ws.setMixedContentMode(WebSettings.MIXED_CONTENT_NEVER_ALLOW);

        webView.addJavascriptInterface(new MiyaBridge(), "Android");

        webView.setWebViewClient(new WebViewClient() {
            @Override
            public WebResourceResponse shouldInterceptRequest(WebView v, WebResourceRequest r) {
                Uri u = r.getUrl();
                if (isOwnOrigin(u)) {
                    WebResourceResponse resp = assetLoader.shouldInterceptRequest(u);
                    if (resp != null) return resp;
                }
                return blocked();
            }

            @Override
            public boolean shouldOverrideUrlLoading(WebView v, WebResourceRequest r) {
                return !isOwnOrigin(r.getUrl()); // bahar ke links block
            }
        });

        webView.setWebChromeClient(new WebChromeClient() {
            @Override
            public void onPermissionRequest(final PermissionRequest req) {
                runOnUiThread(() -> {
                    if (!isOwnOrigin(req.getOrigin())) { req.deny(); return; }
                    handleWebPermission(req, true);
                });
            }

            @Override
            public void onPermissionRequestCanceled(PermissionRequest req) {
                if (pendingWebReq == req) pendingWebReq = null;
            }

            @Override
            public void onGeolocationPermissionsShowPrompt(String origin,
                                                           GeolocationPermissions.Callback cb) {
                boolean allow = origin != null && isOwnOrigin(Uri.parse(origin))
                        && has(Manifest.permission.ACCESS_FINE_LOCATION);
                cb.invoke(origin, allow, false);
            }
        });
    }

    /**
     * Permission na hone par Android permission maangte hain aur milne ke baad WebView ko grant karte hain.
     */
    private void handleWebPermission(PermissionRequest req, boolean mayAsk) {
        List<String> ok = new ArrayList<>();
        List<String> missing = new ArrayList<>();
        for (String r : req.getResources()) {
            if (PermissionRequest.RESOURCE_AUDIO_CAPTURE.equals(r)) {
                if (has(Manifest.permission.RECORD_AUDIO)) ok.add(r);
                else missing.add(Manifest.permission.RECORD_AUDIO);
            } else if (PermissionRequest.RESOURCE_VIDEO_CAPTURE.equals(r)) {
                if (has(Manifest.permission.CAMERA)) ok.add(r);
                else missing.add(Manifest.permission.CAMERA);
            }
        }
        if (!missing.isEmpty() && mayAsk && pendingWebReq == null) {
            pendingWebReq = req;
            ActivityCompat.requestPermissions(this, missing.toArray(new String[0]), REQ_WEB);
            return;
        }
        if (ok.isEmpty()) req.deny(); else req.grant(ok.toArray(new String[0]));
    }

    private static boolean isOwnOrigin(Uri u) {
        return u != null && "https".equals(u.getScheme()) && HOST.equals(u.getHost());
    }

    private static WebResourceResponse blocked() {
        return new WebResourceResponse("text/plain", "utf-8", 403, "Blocked",
                Collections.<String, String>emptyMap(), new ByteArrayInputStream(new byte[0]));
    }

    // ------------------------------------------------------------------
    // PERMISSIONS (ek baar, sirf missing wali, API level ke hisaab se)
    // ------------------------------------------------------------------
    private boolean has(String perm) {
        return ContextCompat.checkSelfPermission(this, perm) == PackageManager.PERMISSION_GRANTED;
    }

    private void requestRuntimePermissions() {
        List<String> need = new ArrayList<>();
        String[] base = {
                Manifest.permission.RECORD_AUDIO,
                Manifest.permission.CAMERA,
                Manifest.permission.ACCESS_FINE_LOCATION
        };
        for (String p : base) if (!has(p)) need.add(p);
        if (Build.VERSION.SDK_INT >= 33 && !has(Manifest.permission.POST_NOTIFICATIONS)) {
            need.add(Manifest.permission.POST_NOTIFICATIONS);
        }
        if (!need.isEmpty()) {
            ActivityCompat.requestPermissions(this, need.toArray(new String[0]), REQ_PERMS);
        }
    }

    @Override
    public void onRequestPermissionsResult(int code, @NonNull String[] perms,
                                           @NonNull int[] results) {
        super.onRequestPermissionsResult(code, perms, results);

        if (code == REQ_WEB) {
            PermissionRequest r = pendingWebReq;
            pendingWebReq = null;
            if (r != null) handleWebPermission(r, false);
            return;
        }

        if (code != REQ_PERMS) return;
        for (int r : results) {
            if (r != PackageManager.PERMISSION_GRANTED) {
                Toast.makeText(this, "Kuch permissions deny hui, kuch features kaam nahi karenge",
                        Toast.LENGTH_LONG).show();
                break;
            }
        }
        // Mic permission ab mili hai -> voice service chalao
        startVoiceService();
    }

    // ------------------------------------------------------------------
    // AUTH (fingerprint / PIN / pattern)
    // ------------------------------------------------------------------
    private void promptAuth() {
        BiometricManager bm = BiometricManager.from(this);
        if (bm.canAuthenticate(AUTHENTICATORS) != BiometricManager.BIOMETRIC_SUCCESS) {
            toast("Phone mein screen lock (PIN/fingerprint) set karo");
            notifyJs(false);
            return;
        }

        BiometricPrompt prompt = new BiometricPrompt(this,
                ContextCompat.getMainExecutor(this),
                new BiometricPrompt.AuthenticationCallback() {
                    @Override
                    public void onAuthenticationSucceeded(
                            @NonNull BiometricPrompt.AuthenticationResult result) {
                        unlocked = true;
                        unlockedAt = SystemClock.elapsedRealtime();
                        toast("🔓 Unlocked");
                        notifyJs(true);
                    }

                    @Override
                    public void onAuthenticationError(int code, @NonNull CharSequence msg) {
                        notifyJs(false);
                    }
                });

        prompt.authenticate(new BiometricPrompt.PromptInfo.Builder()
                .setTitle("Miya unlock")
                .setSubtitle("Apni identity confirm karo")
                .setAllowedAuthenticators(AUTHENTICATORS)
                .build());
    }

    private void notifyJs(boolean ok) {
        if (webView == null) return;
        webView.evaluateJavascript(
                "window.onMiyaAuth && window.onMiyaAuth(" + ok + ")", null);
    }

    private void toast(final String msg) {
        runOnUiThread(() -> Toast.makeText(MainActivity.this, msg, Toast.LENGTH_SHORT).show());
    }

    // ------------------------------------------------------------------
    // PHONE AGENT / TERMUX SETUP (voice se: "Miya agent setup", "Miya termux token save")
    // ------------------------------------------------------------------
    private void openAccessibilitySettings() {
        try {
            startActivity(new Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS));
            toast("Accessibility me 'Miya Agent' ko ON karo");
        } catch (Exception e) {
            Log.w(TAG, "accessibility settings failed", e);
            toast("Settings nahi khul payi");
        }
    }

    /** Termux ka token copy karke ye bolo: "Miya termux token save". Clipboard sirf foreground app padh sakti hai. */
    private void saveTermuxTokenFromClipboard() {
        if (!sessionActive()) { toast("⚠️ Pehle unlock karo"); return; }
        try {
            ClipboardManager cm = (ClipboardManager) getSystemService(Context.CLIPBOARD_SERVICE);
            ClipData d = cm == null ? null : cm.getPrimaryClip();
            String tok = (d != null && d.getItemCount() > 0)
                    ? String.valueOf(d.getItemAt(0).coerceToText(this)).trim() : "";
            if (!TOKEN_PATTERN.matcher(tok).matches()) {
                toast("Clipboard me Termux token nahi hai (32 akshar ka)");
                return;
            }
            MiyaAgent.setTermuxToken(this, tok);
            toast("✅ Termux token save ho gaya");
        } catch (Exception e) {
            Log.w(TAG, "token save failed", e);
            toast("Token save nahi hua");
        }
    }

    /** GitHub token copy karke ye bolo: "Miya github token save". Isse "Miya video banao" workflow chala sakta hai. */
    private void saveGithubTokenFromClipboard() {
        if (!sessionActive()) { toast("⚠️ Pehle unlock karo"); return; }
        try {
            ClipboardManager cm = (ClipboardManager) getSystemService(Context.CLIPBOARD_SERVICE);
            ClipData d = cm == null ? null : cm.getPrimaryClip();
            String tok = (d != null && d.getItemCount() > 0)
                    ? String.valueOf(d.getItemAt(0).coerceToText(this)).trim() : "";
            if (!GH_TOKEN_PATTERN.matcher(tok).matches()) {
                toast("Clipboard me GitHub token nahi hai (github_pat_... se shuru hota hai)");
                return;
            }
            MiyaAgent.setGithubToken(this, tok);
            toast("✅ GitHub token save ho gaya");
        } catch (Exception e) {
            Log.w(TAG, "github token save failed", e);
            toast("Token save nahi hua");
        }
    }

    // ------------------------------------------------------------------
    // JAVASCRIPT BRIDGE (sensitive methods session-gated)
    // JS side: Android.authenticate() ; window.onMiyaAuth = ok => {...}
    // ------------------------------------------------------------------
    public class MiyaBridge {

        @JavascriptInterface
        public void authenticate() {
            runOnUiThread(MainActivity.this::promptAuth);
        }

        @JavascriptInterface
        public boolean isSessionActive() {
            return sessionActive();
        }

        @JavascriptInterface
        public void lock() {
            unlocked = false;
        }

        /** Sirf is app ki screen ki brightness (WRITE_SETTINGS ki zaroorat nahi). */
        @JavascriptInterface
        public void setBrightness(int level) {
            if (!sessionActive()) { toast("⚠️ Pehle unlock karo"); return; }
            final float b = Math.max(0.02f, Math.min(255, Math.max(0, level)) / 255f);
            runOnUiThread(() -> {
                WindowManager.LayoutParams lp = getWindow().getAttributes();
                lp.screenBrightness = b;
                getWindow().setAttributes(lp);
            });
        }

        @JavascriptInterface
        public void openApp(final String packageName) {
            if (!sessionActive()) { toast("⚠️ Pehle unlock karo"); return; }
            if (packageName == null || !PKG_PATTERN.matcher(packageName).matches()) return;
            runOnUiThread(() -> {
                try {
                    Intent i = getPackageManager().getLaunchIntentForPackage(packageName);
                    if (i != null) startActivity(i);
                    else toast("App nahi mili");
                } catch (Exception e) {
                    Log.w(TAG, "openApp failed", e);
                }
            });
        }

        @JavascriptInterface
        public void requestOverlayPermission() {
            if (!sessionActive()) { toast("⚠️ Pehle unlock karo"); return; }
            runOnUiThread(() -> {
                try {
                    if (!Settings.canDrawOverlays(MainActivity.this)) {
                        startActivity(new Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION,
                                Uri.parse("package:" + getPackageName())));
                    }
                } catch (Exception e) {
                    Log.w(TAG, "overlay intent failed", e);
                }
            });
        }

        @JavascriptInterface
        public String getBatteryInfo() {
            try {
                BatteryManager bm = (BatteryManager) getSystemService(BATTERY_SERVICE);
                return new JSONObject()
                        .put("level", bm.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY))

                    String line;

                } catch (SocketTimeoutException slow) {
                    // 25 second se zyada lag raha hai; jitna aaya utna bolo
                }

                if (result.length() > 300) result = result.substring(0, 300);

            } catch (SecurityException auth) {
                closeTermux();
                speak("Termux ne token nahi mana. Token dobara save karo");
            } catch (Exception e) {
                closeTermux();
                speak("Termux se connect nahi ho paya. Termux me listener chalu hai?");
            }
        });

    private static void connectTermux(String token) throws Exception {
        closeTermux();
        Socket s = new Socket();
        s.connect(new InetSocketAddress(TERMUX_HOST, TERMUX_PORT), 3000);
        s.setSoTimeout(25_000);
        tSock = s;
        tOut = s.getOutputStream();
        tIn = new BufferedReader(new InputStreamReader(s.getInputStream(), StandardCharsets.UTF_8));

        // Handshake: pehli line token, server "AUTH_OK" bheje to aage badho
        tOut.write(("TOKEN " + token + "\n").getBytes(StandardCharsets.UTF_8));
        tOut.flush();
        String reply = tIn.readLine();
        if (!"AUTH_OK".equals(reply)) {
            closeTermux();
            throw new SecurityException("auth failed");
        }
    }

                        .put("inputs", new JSONObject().put("mode", mode));
                try (OutputStream os = c.getOutputStream()) {
                    os.write(body.toString().getBytes(StandardCharsets.UTF_8));
                }
                int code = c.getResponseCode();
                if (code == 204) {
                    prefs().edit().putLong(KEY_GH_LAST, now).apply();
                    speak("make".equals(mode)
                            ? "Theek hai, video banna shuru ho gaya. Kuch minute me Telegram par aayegi"
                            : "Comments ke replies shuru kar diye");
                } else if (code == 401 || code == 403) {
                    speak("GitHub ne token nahi mana. Token ki Actions permission aur expiry check karo");
                } else if (code == 404) {
                    speak("Workflow nahi mila. GitHub me automation file hai ya nahi dekho");
                } else if (code == 422) {
                    speak("GitHub ko request samajh nahi aayi. Workflow ka mode input check karo");
                } else {
                    speak("GitHub error " + code);
                }
            } catch (Exception e) {
                speak("GitHub se baat nahi ho payi, internet check karo");
            } finally {
                if (c != null) c.disconnect();
            }
        });
    }
}

package com.miya.app;

import android.accessibilityservice.AccessibilityService;
import android.content.Intent;
import android.os.Bundle;
import android.view.accessibility.AccessibilityEvent;
import android.view.accessibility.AccessibilityNodeInfo;

import java.util.ArrayList;
import java.util.List;
import java.util.regex.Pattern;

/**
 * Miya phone agent ki "aankhein aur haath": screen padhna, tap, type, scroll.
 * Passwords aur lambe numbers (OTP / card jaise) kabhi bahar nahi jaate.
 */
public class MiyaAccessibilityService extends AccessibilityService {

    static volatile MiyaAccessibilityService instance;

    private static final int MAX_NODES = 70;
    private static final Pattern LONG_NUMBER = Pattern.compile("(?:\\d[ -]?){6,}");

    // Pichhli dump ke numbered nodes (CLICK/TYPE inhi ke index se hote hain)
    private final List<AccessibilityNodeInfo> indexed = new ArrayList<>();

    @Override
    protected void onServiceConnected() {
        super.onServiceConnected();
        instance = this;
    }

    @Override
    public boolean onUnbind(Intent intent) {
        instance = null;
        return super.onUnbind(intent);
    }

    @Override
    public void onDestroy() {
        instance = null;
        super.onDestroy();
    }

    @Override public void onAccessibilityEvent(AccessibilityEvent event) { }
    @Override public void onInterrupt() { }

    // ---------- Screen padhna ----------
    synchronized String dumpScreen() {
        indexed.clear();
        AccessibilityNodeInfo root = getRootInActiveWindow();
        if (root == null) return "(screen padh nahi paya)";
        StringBuilder sb = new StringBuilder();
        collect(root, sb);
        if (indexed.isEmpty()) sb.append("(koi padhne layak cheez nahi mili)\n");
        return sb.toString();
    }

    String currentPackage() {
        AccessibilityNodeInfo root = getRootInActiveWindow();
        if (root == null || root.getPackageName() == null) return "";
        return root.getPackageName().toString();
    }

    private void collect(AccessibilityNodeInfo n, StringBuilder sb) {
        if (n == null || indexed.size() >= MAX_NODES) return;
        if (n.isVisibleToUser()) {
            String label = labelOf(n);
            boolean edit = n.isEditable();
            boolean click = n.isClickable();
            if (!label.isEmpty() || edit || click) {
                int idx = indexed.size();
                indexed.add(n);
                sb.append('[').append(idx).append("] ")
                  .append(edit ? "input" : click ? "button" : n.isScrollable() ? "list" : "text");
                if (!label.isEmpty()) sb.append(" \"").append(label).append('"');
                sb.append('\n');
            }
        }
        for (int i = 0; i < n.getChildCount(); i++) collect(n.getChild(i), sb);
    }

    private String labelOf(AccessibilityNodeInfo n) {
        if (n.isPassword()) return "(password field - hidden)";
        CharSequence t = n.getText();
        if (t == null || t.length() == 0) t = n.getContentDescription();
        if (t == null || t.length() == 0) t = n.getHintText();
        if (t == null) return "";
        String s = t.toString().replace('\n', ' ').trim();
        s = LONG_NUMBER.matcher(s).replaceAll("[hidden-number]");
        if (s.length() > 80) s = s.substring(0, 80) + "...";
        return s.replace('"', '\'');
    }

    // ---------- Haath ----------
    private synchronized AccessibilityNodeInfo nodeAt(int i) {
        return (i >= 0 && i < indexed.size()) ? indexed.get(i) : null;
    }

    /** Node ka label (risky-button check ke liye). null = galat index. */
    synchronized String labelAt(int i) {
        AccessibilityNodeInfo n = nodeAt(i);
        return n == null ? null : labelOf(n);
    }

    synchronized boolean click(int i) {
        AccessibilityNodeInfo cur = nodeAt(i);
        for (int depth = 0; depth < 6 && cur != null; depth++) {
            if (cur.isClickable() && cur.performAction(AccessibilityNodeInfo.ACTION_CLICK)) return true;
            cur = cur.getParent();
        }
        return false;
    }

    /** index = -1 ho to jis field par cursor hai wahi. Poora text replace hota hai. */
    synchronized String type(int i, String text) {
        AccessibilityNodeInfo n = nodeAt(i);
        if (n == null && i < 0) {
            AccessibilityNodeInfo root = getRootInActiveWindow();
            if (root != null) n = root.findFocus(AccessibilityNodeInfo.FOCUS_INPUT);
        }
        if (n == null) return "no_field";
        if (n.isPassword()) return "password_field_blocked";
        if (!n.isEditable()) return "not_an_input";
        n.performAction(AccessibilityNodeInfo.ACTION_FOCUS);
        Bundle args = new Bundle();
        args.putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, text);
        return n.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args) ? "typed" : "type_failed";
    }

    synchronized boolean scroll(boolean forward) {
        AccessibilityNodeInfo s = findScrollable(getRootInActiveWindow());
        if (s == null) return false;
        return s.performAction(forward
                ? AccessibilityNodeInfo.ACTION_SCROLL_FORWARD
                : AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD);
    }

    private AccessibilityNodeInfo findScrollable(AccessibilityNodeInfo n) {
        if (n == null) return null;
        if (n.isScrollable() && n.isVisibleToUser()) return n;
        for (int i = 0; i < n.getChildCount(); i++) {
            AccessibilityNodeInfo f = findScrollable(n.getChild(i));
            if (f != null) return f;
        }
        return null;
    }

    /** App kholna. Accessibility service ke context se kholte hain (background se bhi chal jaata hai). */
    boolean launch(Intent intent) {
        try {
            intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
            startActivity(intent);
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    boolean back() { return performGlobalAction(GLOBAL_ACTION_BACK); }
    boolean home() { return performGlobalAction(GLOBAL_ACTION_HOME); }
}
