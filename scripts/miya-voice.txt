package com.miya.app;

import android.Manifest;
import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.app.Service;
import android.content.Intent;
import android.content.pm.PackageManager;
import android.content.pm.ServiceInfo;
import android.os.Build;
import android.os.Bundle;
import android.os.Handler;
import android.os.IBinder;
import android.os.Looper;
import android.speech.RecognitionListener;
import android.speech.RecognizerIntent;
import android.speech.SpeechRecognizer;
import android.util.Log;

import androidx.core.app.NotificationCompat;
import androidx.core.content.ContextCompat;

import java.util.ArrayList;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Foreground (microphone) service: "Miya" wake word sunti hai.
 *  - "Miya agent ..." / "Miya termux ..." -> MiyaAgent sambhalta hai (phone agent / Termux).
 *  - baaki bola hua text MainActivity ko broadcast hota hai (jo use WebView ke JS tak pahunchati hai).
 *  - Agent ne "haan ya nahi" poochha ho to wake word ke bina bhi jawab sun leti hai.
 */
public class MiyaService extends Service {

    private static final String TAG = "MiyaService";

    public static final String ACTION_WAKE = "com.miya.app.WAKE";
    public static final String EXTRA_TEXT = "text";
    private static final String ACTION_STOP = "com.miya.app.STOP_SERVICE";

    private static final String CHANNEL_ID = "miya_channel";
    private static final int NOTIF_ID = 1;

    // Workflow ka "Insert API Token" step isi line ko replace karta hai.
    static final String APP_KEY = "REPLACE_ME";

    private static final Pattern WAKE = Pattern.compile(
            "\\b(miya|mia|maya|meeya)\\b", Pattern.CASE_INSENSITIVE);

    private final Handler handler = new Handler(Looper.getMainLooper());
    private SpeechRecognizer recognizer;
    private boolean inForeground = false;
    private boolean destroyed = false;
    private int errorStreak = 0;

    // ------------------------------------------------------------------
    // LIFECYCLE
    // ------------------------------------------------------------------
    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        if (intent != null && ACTION_STOP.equals(intent.getAction())) {
            stopSelf();
            return START_NOT_STICKY;
        }

        // Android 14+: mic-type FGS ke liye RECORD_AUDIO pehle se honi chahiye
        if (ContextCompat.checkSelfPermission(this, Manifest.permission.RECORD_AUDIO)
                != PackageManager.PERMISSION_GRANTED) {
            Log.w(TAG, "RECORD_AUDIO nahi hai, service band");
            stopSelf();
            return START_NOT_STICKY;
        }

        if (!inForeground && !enterForeground()) {
            stopSelf();
            return START_NOT_STICKY;
        }

        if (!SpeechRecognizer.isRecognitionAvailable(this)) {
            Log.w(TAG, "Speech recognition is device pe available nahi");
            stopSelf();
            return START_NOT_STICKY;
        }

        MiyaAgent.init(this); // bolne (TTS) ke liye
        scheduleListen(0);

        // START_NOT_STICKY: process kill ke baad background se mic-FGS dobara start karna
        // Android 12+ pe ForegroundServiceStartNotAllowedException deta hai (crash)
        return START_NOT_STICKY;
    }

    @Override
    public void onTaskRemoved(Intent rootIntent) {
        stopSelf();
        super.onTaskRemoved(rootIntent);
    }

    @Override
    public void onDestroy() {
        destroyed = true;
        MiyaAgent.onServiceStopped();
        handler.removeCallbacksAndMessages(null);
        if (recognizer != null) {
            try { recognizer.destroy(); } catch (Exception ignored) {}
            recognizer = null;
        }
        if (inForeground) {
            stopForeground(true);
            inForeground = false;
        }
        super.onDestroy();
    }

    @Override
    public IBinder onBind(Intent intent) { return null; }

    // ------------------------------------------------------------------
    // FOREGROUND NOTIFICATION
    // ------------------------------------------------------------------
    private boolean enterForeground() {
        try {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                NotificationChannel ch = new NotificationChannel(
                        CHANNEL_ID, "Miya AI Service", NotificationManager.IMPORTANCE_LOW);
                NotificationManager nm = getSystemService(NotificationManager.class);
                if (nm != null) nm.createNotificationChannel(ch);
            }

            int piFlags = PendingIntent.FLAG_UPDATE_CURRENT
                    | (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M ? PendingIntent.FLAG_IMMUTABLE : 0);
            PendingIntent open = PendingIntent.getActivity(
                    this, 0, new Intent(this, MainActivity.class), piFlags);
            PendingIntent stop = PendingIntent.getService(
                    this, 1, new Intent(this, MiyaService.class).setAction(ACTION_STOP), piFlags);

            Notification n = new NotificationCompat.Builder(this, CHANNEL_ID)
                    .setContentTitle("Miya AI Active")
                    .setContentText("Listening for \"Miya\"...")
                    .setSmallIcon(android.R.drawable.ic_btn_speak_now)
                    .setContentIntent(open)
                    .addAction(0, "Stop", stop)
                    .setOngoing(true)
                    .setPriority(NotificationCompat.PRIORITY_LOW)
                    .build();

            // FOREGROUND_SERVICE_TYPE_MICROPHONE API 30+ mein hai.
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
                startForeground(NOTIF_ID, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_MICROPHONE);
            } else {
                startForeground(NOTIF_ID, n);
            }
            inForeground = true;
            return true;
        } catch (Exception e) {
            Log.w(TAG, "startForeground failed", e);
            return false;
        }
    }

    // ------------------------------------------------------------------
    // WAKE WORD LISTENING (SpeechRecognizer; main thread pe hi chalta hai)
    // ------------------------------------------------------------------
    private void scheduleListen(long delayMs) {
        if (destroyed) return;
        handler.removeCallbacks(listenRunnable);
        handler.postDelayed(listenRunnable, delayMs);
    }

    private final Runnable listenRunnable = new Runnable() {
        @Override public void run() {
            // Miya bol rahi ho to mic band rakho, warna apni hi awaaz sunegi
            if (MiyaAgent.isSpeaking()) { scheduleListen(400); return; }
            startListening();
        }
    };

    private void startListening() {
        if (destroyed) return;
        try {
            if (recognizer == null) {
                recognizer = SpeechRecognizer.createSpeechRecognizer(this);
                recognizer.setRecognitionListener(listener);
            }
            Intent i = new Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH);
            i.putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL,
                    RecognizerIntent.LANGUAGE_MODEL_FREE_FORM);
            i.putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 3);
            i.putExtra(RecognizerIntent.EXTRA_CALLING_PACKAGE, getPackageName());
            recognizer.startListening(i);
        } catch (Exception e) {
            Log.w(TAG, "startListening failed", e);
            onFailure();
        }
    }

    /** Error pe backoff: 0.5s -> 5s tak, taaki loop CPU/battery na khaye. */
    private void onFailure() {
        errorStreak++;
        long delay = Math.min(5000L, 500L * errorStreak);
        scheduleListen(delay);
    }

    private final RecognitionListener listener = new RecognitionListener() {
        @Override public void onReadyForSpeech(Bundle params) {}
        @Override public void onBeginningOfSpeech() {}
        @Override public void onRmsChanged(float rmsdB) {}
        @Override public void onBufferReceived(byte[] buffer) {}
        @Override public void onEndOfSpeech() {}
        @Override public void onPartialResults(Bundle partialResults) {}
        @Override public void onEvent(int eventType, Bundle params) {}

        @Override
        public void onError(int error) {
            switch (error) {
                case SpeechRecognizer.ERROR_INSUFFICIENT_PERMISSIONS:
                    Log.w(TAG, "Mic permission gayi, service band");
                    stopSelf();
                    return;
                case SpeechRecognizer.ERROR_NO_MATCH:
                case SpeechRecognizer.ERROR_SPEECH_TIMEOUT:
                    // Normal: kuch bola nahi gaya. Jaldi dobara sunna shuru.
                    errorStreak = 0;
                    scheduleListen(300);
                    return;
                case SpeechRecognizer.ERROR_RECOGNIZER_BUSY:
                    try { if (recognizer != null) recognizer.cancel(); } catch (Exception ignored) {}
                    onFailure();
                    return;
                default:
                    Log.w(TAG, "Recognizer error " + error);
                    onFailure();
            }
        }

        @Override
        public void onResults(Bundle results) {
            errorStreak = 0;
            ArrayList<String> list =
                    results.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION);
            if (list != null) {
                for (String text : list) {
                    if (text == null) continue;

                    // Agent "haan ya nahi" ka intezaar kar raha hai: wake word ke bina bhi suno
                    if (MiyaAgent.isAwaitingConfirm()) {
                        MiyaAgent.onConfirmText(text);
                        break;
                    }

                    Matcher m = WAKE.matcher(text);
                    if (m.find()) {
                        String command = text.substring(m.end()).trim();
                        // agent / termux commands yahin sambhal liye jaate hain, baaki JS ko
                        if (!MiyaAgent.handleVoice(MiyaService.this, command)) {
                            broadcastWake(command);
                        }
                        break;
                    }
                }
            }
            scheduleListen(300);
        }
    };

    private void broadcastWake(String command) {
        // setPackage: sirf apni app ko jata hai
        Intent b = new Intent(ACTION_WAKE)
                .setPackage(getPackageName())
                .putExtra(EXTRA_TEXT, command);
        sendBroadcast(b);
    }
}
